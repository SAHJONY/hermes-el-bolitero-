// 🎉🎆 Confeti + fuegos artificiales en pantalla cuando un cliente gana (entretenimiento)
function fireConfetti(){
  try{
    if(window.matchMedia&&window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    const c=document.createElement("canvas");
    c.style.cssText="position:fixed;inset:0;pointer-events:none;z-index:99998";
    c.width=window.innerWidth; c.height=window.innerHeight;
    document.body.appendChild(c);
    const ctx=c.getContext("2d");
    const colors=["#FFB838","#38D6BA","#F2545B","#9D8CFF","#FFE9A8","#FFFFFF","#E08C00"];
    const pick=()=>colors[(Math.random()*colors.length)|0];
    const parts=[];
    // lluvia de confeti
    for(let i=0;i<140;i++){ parts.push({fw:false,x:Math.random()*c.width,y:-20-Math.random()*c.height*0.4,r:4+Math.random()*7,col:pick(),vx:-2.5+Math.random()*5,vy:2+Math.random()*4.5,rot:Math.random()*6.28,vr:-0.25+Math.random()*0.5}); }
    // fuegos artificiales (estallidos radiales escalonados)
    for(let b=0;b<6;b++){
      const cx=c.width*(0.18+Math.random()*0.64), cy=c.height*(0.16+Math.random()*0.34), col=pick(), delay=b*360;
      for(let i=0;i<38;i++){ const a=(Math.PI*2*i)/38, sp=2+Math.random()*4.2;
        parts.push({fw:true,x:cx,y:cy,r:2+Math.random()*2.6,col,vx:Math.cos(a)*sp,vy:Math.sin(a)*sp,rot:0,vr:0,delay,life:1}); }
    }
    const start=performance.now();
    let done=false;
    const cleanup=()=>{ if(done) return; done=true; try{ c.remove(); }catch(e){} };
    (function frame(now){
      if(done) return;
      const el=now-start; ctx.clearRect(0,0,c.width,c.height);
      for(const p of parts){
        if(p.fw){
          if(el<p.delay) continue;
          p.x+=p.vx; p.y+=p.vy; p.vy+=0.03; p.vx*=0.99; p.vy*=0.99; p.life-=0.012;
          if(p.life<=0) continue;
          ctx.save(); ctx.globalAlpha=Math.max(0,p.life); ctx.fillStyle=p.col; ctx.beginPath(); ctx.arc(p.x,p.y,p.r,0,6.283); ctx.fill(); ctx.restore();
        } else {
          p.x+=p.vx; p.y+=p.vy; p.vy+=0.06; p.rot+=p.vr;
          ctx.save(); ctx.translate(p.x,p.y); ctx.rotate(p.rot); ctx.fillStyle=p.col; ctx.fillRect(-p.r/2,-p.r/2,p.r,p.r*0.62); ctx.restore();
        }
      }
      if(el<3600) requestAnimationFrame(frame); else cleanup();
    })(start);
    // Failsafe: limpia el canvas aunque rAF esté pausado (pestaña oculta).
    setTimeout(cleanup,5000);
  }catch(e){}
}

/* ============================ JUEGO: RASPADITO DE LA SUERTE ============================
   Tarjeta de raspar 3×3 con un juego gratis al día. El cartón se genera y se
   persiste en la primera raspada (no se puede "re-rollear" recargando). Si hay
   tres símbolos iguales, paga el premio del símbolo vía addCoins(). Solo CSS. */
const RASP_SYMS=[{e:"🪙",v:50,w:40},{e:"🥭",v:75,w:30},{e:"🍹",v:150,w:18},{e:"⭐",v:300,w:9},{e:"💎",v:500,w:3}];
function pickRaspSym(rnd){ // elige símbolo ponderado hacia los de menor valor
  const tot=RASP_SYMS.reduce((a,s)=>a+s.w,0); let r=rnd()*tot;
  for(const s of RASP_SYMS){ if((r-=s.w)<0) return s; } return RASP_SYMS[0];
}
function genRaspCard(rnd=Math.random){
  const win=rnd()<0.4;            // ~40% de cartones premiados
  const cells=new Array(9).fill(null);
  const counts={};
  if(win){
    const wsym=pickRaspSym(rnd);                 // símbolo ganador
    const order=[0,1,2,3,4,5,6,7,8];
    for(let i=order.length-1;i>0;i--){ const j=Math.floor(rnd()*(i+1)); [order[i],order[j]]=[order[j],order[i]]; }
    order.slice(0,3).forEach(p=>cells[p]=wsym.e); // coloca exactamente 3
    for(let i=0;i<9;i++){ if(cells[i])continue;
      let s; do{ s=pickRaspSym(rnd); }while(s.e===wsym.e||(counts[s.e]||0)>=2); // evita otro trío y más del ganador
      cells[i]=s.e; counts[s.e]=(counts[s.e]||0)+1; }
    return {cells,prize:wsym.v,winSym:wsym.e};
  }
  for(let i=0;i<9;i++){ let s; do{ s=pickRaspSym(rnd); }while((counts[s.e]||0)>=2); // ningún símbolo llega a 3
    cells[i]=s.e; counts[s.e]=(counts[s.e]||0)+1; }
  return {cells,prize:0,winSym:null};
}
function Raspadito(){
  useLang();
  const [today]=useState(dateKey(new Date()));
  const [card,setCard]=useState(null);            // {cells,prize,winSym}
  const [revealed,setRevealed]=useState([]);      // índices destapados
  const [done,setDone]=useState(false);           // todas destapadas + premio cobrado
  const awardedRef=useRef(false);
  useEffect(()=>{(async()=>{ const r=await sGet("rasp:day",false);
    if(r&&r.day===today){ setCard({cells:r.cells,prize:r.prize,winSym:r.winSym}); setRevealed(r.revealed||[]); if(r.done){ setDone(true); awardedRef.current=true; } }
  })();},[today]);
  const ensureCard=async()=>{ if(card) return card; const c=genRaspCard();
    await sSet("rasp:day",{day:today,cells:c.cells,prize:c.prize,winSym:c.winSym,revealed:[],done:false},false);
    setCard(c); return c; };
  const persist=(c,rev,fin)=>sSet("rasp:day",{day:today,cells:c.cells,prize:c.prize,winSym:c.winSym,revealed:rev,done:fin},false);
  const finish=async(c,rev)=>{ setDone(true);
    if(c.prize>0&&!awardedRef.current){ awardedRef.current=true; try{ await addCoins(c.prize); }catch(e){} fireConfetti(); }
    await persist(c,rev,true); };
  const reveal=async(i)=>{ if(done||revealed.includes(i)) return; const c=await ensureCard();
    const rev=[...revealed,i]; setRevealed(rev);
    if(rev.length>=9) await finish(c,rev); else await persist(c,rev,false); };
  const revealAll=async()=>{ if(done) return; const c=await ensureCard(); const rev=[0,1,2,3,4,5,6,7,8]; setRevealed(rev); await finish(c,rev); };
  return (
    <div className="card" style={{marginTop:14,textAlign:"center"}}>
      <div className="disp" style={{fontWeight:800,fontSize:16,marginBottom:4}}>{t("rasp.title")}</div>
      <div style={{fontSize:12,color:T.dim,marginBottom:12}}>{t("rasp.sub")}</div>
      <div style={{display:"grid",gridTemplateColumns:"repeat(3,1fr)",gap:8,maxWidth:240,margin:"0 auto 12px"}}>
        {[0,1,2,3,4,5,6,7,8].map(i=>{ const up=revealed.includes(i); const win=done&&card&&card.winSym&&card.cells[i]===card.winSym;
          return (
            <button key={i} aria-label={up?(card?card.cells[i]:""):t("rasp.hint")} onClick={()=>reveal(i)} disabled={done&&!up}
              style={{aspectRatio:"1",borderRadius:12,fontSize:30,lineHeight:1,cursor:up?"default":"pointer",
                border:win?"2px solid "+T.teal:"1px solid rgba(255,255,255,.08)",
                background:up?(win?"rgba(56,214,186,.16)":T.raise):"linear-gradient(135deg,#C9A24B,#8A6D2F)",
                color:up?"#fff":"rgba(255,255,255,.85)",fontWeight:800,
                boxShadow:up?"none":"inset 0 1px 2px rgba(255,255,255,.25),0 2px 5px rgba(0,0,0,.35)",
                transition:"background .15s, transform .12s",transform:up?"none":"scale(1)"}}>
              {up?card?.cells[i]:"❓"}
            </button>
          );
        })}
      </div>
      {!done&&<button className="btn primary" style={{width:"100%"}} onClick={revealAll}>{t("rasp.reveal_all")}</button>}
      {!done&&<div style={{fontSize:11,color:T.dim,marginTop:8}}>{t("rasp.hint")}</div>}
      {done&&(
        <div className="card" style={{background:T.raise,marginTop:4}}>
          {card&&card.prize>0
            ? <div className="disp" style={{fontWeight:800,color:T.teal}}>{t("rasp.won")} {card.winSym} {t("rasp.won2")} {card.prize} 🪙! 🎉</div>
            : <div style={{color:T.dim,fontSize:13}}>{t("rasp.lose")}</div>}
          <div style={{fontSize:11,color:T.dim,marginTop:6}}>{t("rasp.come_back")}</div>
        </div>
      )}
    </div>
  );
}

/* ============================ ARCADE: HELPERS COMPARTIDOS ============================
   useDaily(key): un juego gratis al día, persistido (no se puede "re-rollear"
   recargando). saved=null → jugable · saved=obj → resultado del día · loading
   mientras carga. awardCoins paga + confeti. GameShell/GameResult: chrome común. */
function useDaily(key){
  const [today]=useState(dateKey(new Date()));
  const [saved,setSaved]=useState(null);
  const [loading,setLoading]=useState(true);
  useEffect(()=>{(async()=>{ try{ const r=await sGet(key+":day",false); if(r&&r.day===today) setSaved(r); }catch(e){} setLoading(false); })();},[key,today]);
  const save=async(data)=>{ const rec={day:today,...data}; try{ await sSet(key+":day",rec,false); }catch(e){} setSaved(rec); };
  return {saved,loading,save};
}

/* ---------------- TRAGAMONEDAS HABANA (slots de 3 carretes) ---------------- */
const SLOT_SYMS=[{e:"🍒",v:100,w:34},{e:"🔔",v:150,w:26},{e:"🍀",v:200,w:18},{e:"⭐",v:300,w:12},{e:"💎",v:500,w:7},{e:"7️⃣",v:777,w:3}];
function pickSlot(rnd){ const tot=SLOT_SYMS.reduce((a,s)=>a+s.w,0); let r=rnd()*tot; for(const s of SLOT_SYMS){ if((r-=s.w)<0) return s; } return SLOT_SYMS[0]; }
function genSlots(rnd=Math.random){
  const r=rnd();
  if(r<0.16){ const s=pickSlot(rnd); return {reels:[s.e,s.e,s.e],prize:s.v,kind:3}; }      // 3 iguales
  if(r<0.42){                                                                                // 2 iguales
    const a=pickSlot(rnd); let b; do{ b=pickSlot(rnd); }while(b.e===a.e);
    const reels=[a.e,a.e,b.e];
    for(let i=reels.length-1;i>0;i--){ const j=Math.floor(rnd()*(i+1)); [reels[i],reels[j]]=[reels[j],reels[i]]; }
    return {reels,prize:40,kind:2};
  }
  const set=new Set(),reels=[];                                                              // 3 distintos
  while(reels.length<3){ const s=pickSlot(rnd); if(set.has(s.e))continue; set.add(s.e); reels.push(s.e); }
  return {reels,prize:0,kind:0};
}

  };
  return (

function HiLo(){

  const [cur,setCur]=useState(null); const [pot,setPot]=useState(0); const [playing,setPlaying]=useState(false); const [flash,setFlash]=useState("");

  const start=()=>{ if(loading||saved) return; setCur(rnd()); setPot(0); setPlaying(true); setFlash(""); };
  const guess=async(dir)=>{
    if(!playing) return; const next=rnd();
    if(next===cur){ setFlash(t("hilo.push")); setCur(next); return; }
    const correct=(dir==="hi"&&next>cur)||(dir==="lo"&&next<cur);
    if(correct){ setPot(p=>p+25); setCur(next); setFlash("+25 🪙"); }

/* ---------------- VEINTIUNO (21) — blackjack-lite vs crupier ---------------- */
const BJ_SUITS=["♠","♥","♦","♣"];
function bjDraw(){ return {r:1+Math.floor(Math.random()*13),s:BJ_SUITS[Math.floor(Math.random()*4)]}; }
function bjLabel(c){ return ({1:"A",11:"J",12:"Q",13:"K"})[c.r]||String(c.r); }
function bjTotal(hand){ let sum=0,aces=0; for(const c of hand){ if(c.r===1){ aces++; sum+=11; } else sum+=Math.min(c.r,10); } while(sum>21&&aces){ sum-=10; aces--; } return sum; }
function BlackjackLite(){
  useLang(); const {saved,loading,save}=useDaily("bj");
  const [ph,setPh]=useState(null); const [dh,setDh]=useState(null); const [phase,setPhase]=useState("idle"); // idle | player | done
  const finish=async(p,d,natural)=>{
    setPh(p); setDh(d); setPhase("done");
    const pt=bjTotal(p),dt=bjTotal(d); let outcome,prize=0;
    if(pt>21) outcome="bust";
    else if(natural){ outcome="bj"; prize=300; }
    else if(dt>21||pt>dt){ outcome="win"; prize=200; }
    else if(pt===dt) outcome="push";
    else outcome="lose";
    if(prize>0) await awardCoins(prize);
    await save({p,d,pt,dt,outcome,prize});
  };
  const deal=()=>{ if(loading||saved) return; const p=[bjDraw(),bjDraw()],d=[bjDraw(),bjDraw()]; setPh(p); setDh(d);
    if(bjTotal(p)===21) finish(p,d,true); else setPhase("player"); };
  const hit=()=>{ const p=[...ph,bjDraw()]; setPh(p); if(bjTotal(p)>21) finish(p,dh,false); };
  const stand=()=>{ const d=[...dh]; while(bjTotal(d)<17) d.push(bjDraw()); finish(ph,d,false); };
  const S=saved||{}; const showP=saved?S.p:ph; const showD=saved?S.d:dh;
  const chip=(c,hidden,key)=>(
    <span key={key} style={{display:"inline-block",minWidth:26,padding:"6px 5px",margin:2,borderRadius:7,
      background:hidden?"#16435a":"#fff",color:hidden?"#16435a":((c.s==="♥"||c.s==="♦")?"#C0392B":"#1a2530"),
      fontWeight:800,fontSize:15}}>{hidden?"🂠":bjLabel(c)+c.s}</span>
  );
  const dealerHidden=(i)=> phase==="player"&&!saved&&i>=1;
  return (
    <GameShell title={t("bj.title")} sub={t("bj.sub")}>
      {(phase!=="idle"||saved)&&(
        <div style={{textAlign:"left",margin:"4px 0 12px"}}>
          <div style={{fontSize:12,color:T.dim}}>{t("bj.dealer")}{(saved||phase==="done")?(" · "+S.dt):""}</div>
          <div>{showD&&showD.map((c,i)=>chip(c,dealerHidden(i),i))}</div>
          <div style={{fontSize:12,color:T.dim,marginTop:8}}>{t("bj.you")} · {bjTotal(showP||[])}</div>
          <div>{showP&&showP.map((c,i)=>chip(c,false,i))}</div>
        </div>
      )}
      {!saved&&phase==="idle"&&<button className="btn primary" style={{width:"100%"}} disabled={loading} onClick={deal}>{t("bj.deal")}</button>}
      {!saved&&phase==="player"&&(
        <div style={{display:"flex",gap:8}}>
          <button className="btn primary" style={{flex:1}} onClick={hit}>{t("bj.hit")}</button>
          <button className="btn ghost" style={{flex:1}} onClick={stand}>{t("bj.stand")}</button>
        </div>
      )}
      {saved&&<GameResult won={S.prize>0}>{
        S.outcome==="bj"?t("bj.bj"):
        S.outcome==="win"?tf("bj.win",{n:S.prize}):
        S.outcome==="push"?t("bj.push"):
        S.outcome==="bust"?t("bj.bust"):t("bj.lose")
      }</GameResult>}
    </GameShell>
  );
}

/* ---------------- MEMORIA CUBANA (memory match, 4 parejas) ---------------- */
const MEM_ICONS=["🍒","🔔","🍀","⭐","💎","🎰","🪙","🎲"];
function MemoryGame(){
  useLang(); const {saved,loading,save}=useDaily("mem");
  const [deck,setDeck]=useState(null); const [flipped,setFlipped]=useState([]); const [matched,setMatched]=useState([]); const [moves,setMoves]=useState(0); const [lock,setLock]=useState(false);
  const shuffle=(arr)=>{ for(let i=arr.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [arr[i],arr[j]]=[arr[j],arr[i]]; } return arr; };
  const start=()=>{ if(loading||saved) return;
    const four=shuffle([...MEM_ICONS]).slice(0,4);
    setDeck(shuffle([...four,...four])); setFlipped([]); setMatched([]); setMoves(0); setLock(false); };
  const finish=async(mv)=>{ const prize=Math.max(60,260-(mv-4)*25); await awardCoins(prize); await save({moves:mv,prize}); };
  const flip=(i)=>{
    if(lock||!deck||flipped.includes(i)||matched.includes(i)) return;
    const f=[...flipped,i]; setFlipped(f);
    if(f.length===2){ const mv=moves+1; setMoves(mv); setLock(true); const [a,b]=f;
      if(deck[a]===deck[b]){ const nm=[...matched,a,b]; setMatched(nm); setFlipped([]); setLock(false); if(nm.length===8) finish(mv); }
      else setTimeout(()=>{ setFlipped([]); setLock(false); },750);
    }
  };
  const isUp=(i)=> flipped.includes(i)||matched.includes(i);
  return (
    <GameShell title={t("mem.title")} sub={t("mem.sub")}>
      {deck&&!saved&&<div style={{fontSize:12,color:T.dim,marginBottom:8}}>{t("mem.moves")}: <b>{moves}</b></div>}
      {!saved&&!deck&&<button className="btn primary" style={{width:"100%"}} disabled={loading} onClick={start}>{t("hilo.start")}</button>}
      {deck&&!saved&&(
        <div style={{display:"grid",gridTemplateColumns:"repeat(4,1fr)",gap:8,maxWidth:260,margin:"0 auto"}}>
          {deck.map((e,i)=>{ const up=isUp(i); return (
            <button key={i} onClick={()=>flip(i)} disabled={up}
              style={{aspectRatio:"1",borderRadius:10,fontSize:26,fontWeight:800,border:"1px solid rgba(255,255,255,.08)",
                background:up?T.raise:"linear-gradient(135deg,#1f6f8b,#16435a)",color:"#fff"}}>{up?e:"❓"}</button>
          );})}
        </div>
      )}
      {saved&&<GameResult won>{tf("mem.win",{n:saved.prize})} · {t("mem.moves")}: {saved.moves}</GameResult>}
    </GameShell>
  );
}
