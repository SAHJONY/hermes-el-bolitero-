function TabAlertas(){
  const TIROS=["Mañana","Mediodía","Tarde","Noche"];
  const [f,setF]=useState({nombre:"",tel:"",email:"",voz:true,mail:true,tg:true,tiros:{Mañana:false,"Mediodía":true,Tarde:false,Noche:true},activo:true});
  const [estado,setEstado]=useState("");
  const [yaInscrito,setYa]=useState(false);
  const [tgLink,setTgLink]=useState("");
  useEffect(()=>{(async()=>{
    const mine=await sGet("alerta:mine",false);
    if(mine){ setF(x=>({...x,...mine.data})); setYa(true); }
    const tg=await sGet("canal:telegram",true); if(tg) setTgLink(tg);
  })();},[]);
  const guardar=async()=>{
    if(!f.tel.trim()&&!f.email.trim()){ setEstado("Pon al menos un teléfono o un email."); return; }
    if(f.voz&&!f.tel.trim()){ setEstado("Para la llamada de Hermes hace falta tu teléfono."); return; }
    let mine=await sGet("alerta:mine",false);
    const id=mine?.id||("s"+Math.random().toString(16).slice(2,10));
    const data={...f,ts:Date.now()};
    const subs=await sGet("alertas:subs",true)||{};
    subs[id]=data;
    const ok=await sSet("alertas:subs",subs,true);
    await sSet("alerta:mine",{id,data},false);
    setYa(true);
    setEstado(ok?"✔ Inscrito. Cuando caiga el número, Hermes te avisa.":"No se pudo guardar — intenta de nuevo.");
  };
  const baja=async()=>{
    const mine=await sGet("alerta:mine",false);
    if(mine){ const subs=await sGet("alertas:subs",true)||{}; delete subs[mine.id]; await sSet("alertas:subs",subs,true); }
    try{ await window.storage.delete("alerta:mine"); }catch{}
    setYa(false); setEstado("Te diste de baja de las alertas.");
  };
  return (
    <div className="fade-in">
      <Section title="Alertas de Hermes 🔔" sub="Cuando caiga el número, Hermes te llama, te escribe al correo o lo canta en el grupo de Telegram">
        <div className="card" style={{display:"grid",gap:10}}>
          <input className="input" placeholder="Tu nombre" value={f.nombre} onChange={e=>setF({...f,nombre:e.target.value})}/>
          <input className="input mono" placeholder="Teléfono (+1 786 …)" inputMode="tel" value={f.tel} onChange={e=>setF({...f,tel:e.target.value})}/>
          <input className="input mono" placeholder="Email" inputMode="email" value={f.email} onChange={e=>setF({...f,email:e.target.value})}/>
          <div>
            <Label>¿Cómo te aviso?</Label>
            <div style={{display:"flex",gap:8}}>
              <button className={"chip"+(f.voz?" on":"")} style={{flex:1,justifyContent:"center"}} onClick={()=>setF({...f,voz:!f.voz})}>📞 Llamada</button>
              <button className={"chip"+(f.mail?" on":"")} style={{flex:1,justifyContent:"center"}} onClick={()=>setF({...f,mail:!f.mail})}>✉️ Email</button>
              <button className={"chip"+(f.tg?" on":"")} style={{flex:1,justifyContent:"center"}} onClick={()=>setF({...f,tg:!f.tg})}>📣 Telegram</button>
            </div>
          </div>
          <div>
            <Label>¿Qué tiros?</Label>
            <div style={{display:"flex",gap:6,flexWrap:"wrap"}}>
              {TIROS.map(t=>(
                <button key={t} className={"chip"+(f.tiros[t]?" on":"")} onClick={()=>setF({...f,tiros:{...f.tiros,[t]:!f.tiros[t]}})}>{t}</button>
              ))}
            </div>
          </div>
          <button className="btn primary" onClick={guardar}>{yaInscrito?"Actualizar mis alertas":"🔔 Quiero que Hermes me avise"}</button>
          {yaInscrito&&<button className="btn ghost" onClick={baja}>Darme de baja</button>}
          {estado&&<div style={{fontSize:12,color:estado.startsWith("✔")?T.teal:T.coral}}>{estado}</div>}
        </div>
      </Section>
      {tgLink&&(
        <div className="card" style={{marginBottom:12,textAlign:"center"}}>
          <div className="disp" style={{fontWeight:800,fontSize:15,marginBottom:6}}>📣 Grupo oficial · HERMES EL BOLITERO</div>
          <div style={{fontSize:12,color:T.dim,marginBottom:10}}>Los números se cantan en vivo en el grupo de Telegram.</div>
          <a href={tgLink} target="_blank" rel="noreferrer" className="btn primary" style={{display:"block",textAlign:"center",textDecoration:"none"}}>Unirme al grupo de Telegram</a>
        </div>
      )}
      <div style={{fontSize:11,color:T.dim,lineHeight:1.6}}>
        Tu contacto se guarda en el registro de alertas de la comunidad para que el dueño y su sistema
        puedan avisarte; no se usa para otra cosa y puedes borrarte cuando quieras. Las alertas son
        informativas — Hermes canta el número, nunca gestiona jugadas.
      </div>
    </div>
  );
}


/* ============================ TAB: MI TICKET (souvenir) ============================ */
function TabTicket(){
  const [board,setBoard]=useState("labolita");
  const lottery=LOTTERIES.find(l=>l.id===board);
  const [sess,setSess]=useState("Noche");
  const [fijo,setFijo]=useState("");
  const [c1,setC1]=useState(""); const [c2,setC2]=useState("");
  const [monto,setMonto]=useState("");
  const [tickets,setTickets]=useState(null);
  const PAYOUTS={fijo:75,corrido:25}; // multiplicadores referenciales (solo información)
  const num=v=>{ const n=parseFloat(String(v||"").replace(/[^0-9.]/g,"")); return isNaN(n)?0:n; };
  const nCorr=[c1,c2].filter(Boolean).length;
  const estimado=num(monto)*(PAYOUTS.fijo + nCorr*PAYOUTS.corrido);
  const [msg,setMsg]=useState("");
  const [checking,setChecking]=useState(false);
  const d2ok=v=>/^\d{2}$/.test(v);
  useEffect(()=>{(async()=>{ setTickets(await sGet("tickets:mine",false)||[]); })();},[]);
  useEffect(()=>{ if(!lottery.sessions.includes(sess)) setSess(lottery.sessions[lottery.sessions.length-1]); },[board]);
  const save=async(list)=>{ setTickets(list); await sSet("tickets:mine",list,false);
    const all=await sGet("tickets:all",true)||{}; let mine=await sGet("alerta:mine",false);
    all[(mine?.id)||"anon-"+(list[0]?.id||"x")]=list.slice(0,20); await sSet("tickets:all",all,true); };
  const crear=async()=>{
    setMsg("");
    if(!d2ok(fijo)){ setMsg("El fijo son 2 dígitos (00–99)."); return; }
    if((c1&&!d2ok(c1))||(c2&&!d2ok(c2))){ setMsg("Los corridos son de 2 dígitos."); return; }
    const t={ id:"HB-"+Math.random().toString(36).slice(2,8).toUpperCase(),
      board, boardName:lottery.short.replace(" 🔴",""), session:sess, date:dateKey(new Date()),
      fijo, corridos:[c1,c2].filter(Boolean), monto:monto.trim(), ts:Date.now(), status:"pendiente", premio:null };
    await save([t,...(tickets||[])].slice(0,30));
    setFijo("");setC1("");setC2("");setMonto("");
    setMsg("🎟️ Ticket creado — guárdalo y juega esos mismos números en una tienda autorizada.");
  };
  const comprobar=async()=>{
    if(checking) return; setChecking(true); setMsg("");
    try{
      const live=await fetchLiveDraws();
      let wins=0;
      const next=(tickets||[]).map(t=>{
        if(t.status!=="pendiente") return t;
        const d=pickDraw(live.draws,t.board,t.date,t.session);
        if(!d) return t;
        const f2=String(d.pick3).padStart(3,"0").slice(1);
        const ca=String(d.pick4).padStart(4,"0").slice(0,2), cb=String(d.pick4).padStart(4,"0").slice(2,4);
        const hits=[];
        if(t.fijo===f2) hits.push("FIJO "+f2);
        t.corridos.forEach(c=>{ if(c===ca||c===cb) hits.push("corrido "+c); });
        return {...t, status:hits.length?"ganador":"no salió", premio:hits.join(" + ")||null, resultado:f2+" · "+ca+" "+cb};
      });
      wins=next.filter(t=>t.status==="ganador").length-(tickets||[]).filter(t=>t.status==="ganador").length;
      await save(next);
      setMsg(wins>0?("🎉 ¡¡TU NÚMERO SALIÓ!! "+wins+" ticket(s) ganador(es) — Hermes también te avisará por tus canales de Alertas."):"Comprobado contra los tiros oficiales publicados.");
    }catch{ setMsg("La señal no respondió — intenta en un momento."); }
    setChecking(false);
  };
  const compartir=async(t)=>{
    const txt=`🎟️ HERMES EL BOLITERO · Ticket ${t.id}\n${t.boardName} · ${t.session} · ${t.date}\nFIJO ${t.fijo}${t.corridos.length?" · corridos "+t.corridos.join(" "):""}\nRecuerdo sin valor — juego estos números en la lotería legal.`;
    try{ if(navigator.share) await navigator.share({text:txt}); else { await navigator.clipboard.writeText(txt); setMsg("Ticket copiado ✔"); } }catch{}
  };
  const ganados=(tickets||[]).filter(t=>t.status==="ganador").length;
  return (
    <div className="fade-in">
      {ganados>0&&(
        <div className="card" style={{marginBottom:12,border:"1.5px solid #FFD978",background:"linear-gradient(135deg, rgba(255,184,56,.18), rgba(19,34,43,.9))",textAlign:"center"}}>
          <div style={{fontSize:22}}>🎉👑🎉</div>
          <div className="disp" style={{fontWeight:800,color:"#FFD978"}}>¡Tienes {ganados} ticket(s) con número ganador!</div>
          <div style={{fontSize:11,color:T.dim,marginTop:4}}>Si lo jugaste en la tienda legal, ve a cobrar allá. Aquí celebramos contigo 🥳</div>
        </div>
      )}
      <Section title="Elige tus números 🎟️" sub="Crea tu ticket souvenir — sin dinero: lo juegas tú en la lotería legal">
        <div className="card" style={{display:"grid",gap:10}}>
          <div className="scroll-x">{LOTTERIES.map(l=><button key={l.id} className={"chip"+(board===l.id?" on":"")} onClick={()=>setBoard(l.id)}>{l.short}</button>)}</div>
          <div className="scroll-x">{lottery.sessions.map(s=><button key={s} className={"chip"+(sess===s?" on":"")} onClick={()=>setSess(s)}>{s}</button>)}</div>
          <div style={{display:"flex",gap:8}}>
            <div style={{flex:1}}><Label>Fijo</Label>
              <input className="input mono" inputMode="numeric" maxLength={2} placeholder="17" value={fijo} onChange={e=>setFijo(e.target.value.replace(/\D/g,""))} style={{textAlign:"center",fontSize:22,fontWeight:700}}/></div>
            <div style={{flex:1}}><Label>Corrido 1</Label>
              <input className="input mono" inputMode="numeric" maxLength={2} placeholder="—" value={c1} onChange={e=>setC1(e.target.value.replace(/\D/g,""))} style={{textAlign:"center",fontSize:22}}/></div>
            <div style={{flex:1}}><Label>Corrido 2</Label>
              <input className="input mono" inputMode="numeric" maxLength={2} placeholder="—" value={c2} onChange={e=>setC2(e.target.value.replace(/\D/g,""))} style={{textAlign:"center",fontSize:22}}/></div>
          </div>
          {fijo.length===2&&<div style={{fontSize:12,color:T.mango,textAlign:"center"}}>{parseInt(fijo,10)===0?100:parseInt(fijo,10)} · {charadaOf(parseInt(fijo,10))}</div>}
          <div>
            <Label>Monto que juegas (calculadora) 🧮</Label>
            <input className="input mono" inputMode="decimal" placeholder="$ ej: 5" value={monto} onChange={e=>setMonto(e.target.value)} style={{textAlign:"center",fontSize:18}}/>
          </div>
          {num(monto)>0&&(
            <div className="card" style={{background:T.raise,textAlign:"center",padding:"10px 12px"}}>
              <div style={{fontSize:11,color:T.dim}}>Ganancia estimada si sale tu número (referencial)</div>
              <div className="disp" style={{fontWeight:800,fontSize:22,color:T.teal}}>≈ ${estimado.toFixed(2)}</div>
              <div style={{fontSize:10,color:T.dim}}>Fijo ×{PAYOUTS.fijo}{nCorr>0?` · ${nCorr} corrido(s) ×${PAYOUTS.corrido}`:""} · solo información, no es pago garantizado</div>
            </div>
          )}
          <button className="btn primary" onClick={crear}>🎟️ Generar mi ticket souvenir</button>
          <button className="btn ghost" onClick={comprobar} disabled={checking}>{checking?"Comprobando con los tiros oficiales…":"🔎 Comprobar mis tickets"}</button>
          {msg&&<div style={{fontSize:12,color:msg.includes("🎉")||msg.includes("✔")||msg.includes("🎟️")?T.teal:T.coral}}>{msg}</div>}
        </div>
      </Section>
      <Section title="Mis tickets" right={<span style={{fontSize:11,color:T.dim}}>{(tickets||[]).length}</span>}>
        <div style={{display:"grid",gap:10}}>
          {tickets&&tickets.length===0&&<div className="card" style={{color:T.dim,fontSize:13,textAlign:"center"}}>Aún no tienes tickets. Elige tus números arriba.</div>}
          {(tickets||[]).map(t=>(
            <div key={t.id} className="card" style={{padding:0,overflow:"hidden",border:t.status==="ganador"?"1.5px solid #FFD978":`1px solid ${T.line}`}}>
              <div style={{background:"linear-gradient(135deg,#FFE9A8,#FFB838 60%,#E08C00)",color:"#3A2604",padding:"8px 14px",display:"flex",justifyContent:"space-between",alignItems:"center"}}>
                <span className="disp" style={{fontWeight:800,fontSize:13}}>HERMES EL BOLITERO</span>
                <span className="mono" style={{fontSize:11,fontWeight:700}}>{t.id}</span>
              </div>
              <div style={{padding:"12px 14px"}}>
                <div style={{fontSize:11,color:T.dim,marginBottom:8,display:"flex",justifyContent:"space-between"}}>
                  <span>{t.boardName} · {t.session} · {t.date}</span>
                  {num(t.monto)>0&&<span style={{color:T.mango}}>Jugado ${num(t.monto).toFixed(2)} · posible ≈ ${(num(t.monto)*(PAYOUTS.fijo+(t.corridos.length*PAYOUTS.corrido))).toFixed(2)}</span>}
                </div>
                <div style={{display:"flex",gap:10,alignItems:"center"}}>
                  <div className="ball hot" style={{width:54,height:54,fontSize:21}}>{t.fijo}</div>
                  {t.corridos.map((c,i)=><div key={i} className="ball">{c}</div>)}
                  <div style={{marginLeft:"auto",textAlign:"right"}}>
                    {t.status==="pendiente"&&<span className="chip" style={{cursor:"default"}}>⏳ pendiente</span>}
                    {t.status==="ganador"&&<span className="chip" style={{cursor:"default",background:"#FFD978",color:"#3A2604",fontWeight:800}}>🏆 {t.premio}</span>}
                    {t.status==="no salió"&&<span className="chip" style={{cursor:"default"}}>{t.resultado||"no salió"}</span>}
                  </div>
                </div>
                <div style={{borderTop:`1px dashed ${T.line}`,marginTop:12,paddingTop:8,display:"flex",justifyContent:"space-between",alignItems:"center"}}>
                  <span style={{fontSize:9,color:T.dim,maxWidth:"70%"}}>RECUERDO SIN VALOR MONETARIO · Juega estos números en una tienda autorizada de la lotería legal de tu estado. Aquí no se apuesta ni se paga premio.</span>
                  <button className="chip" onClick={()=>compartir(t)}>📤</button>
                </div>
              </div>
            </div>
          ))}
        </div>
      </Section>
      <div style={{fontSize:11,color:T.dim,lineHeight:1.6}}>
        Cuando tu número salga, lo verás aquí en tu cuenta al instante — y si activaste 🔔 Alertas,
        Hermes te avisa también por llamada, email o el grupo de Telegram del dueño.
      </div>
    </div>
  );
}

/* ============================ TAB: CONFIRMACIÓN DE PAGO ============================ */
function TabPago(){
  const [metodos,setMetodos]=useState([]);
  const [metodo,setMetodo]=useState("");
  const [monto,setMonto]=useState("");
  const [tx,setTx]=useState("");
  const [recibo,setRecibo]=useState("");
  const [contacto,setContacto]=useState("");
  const [busy,setBusy]=useState(false);
  const [msg,setMsg]=useState("");
  const [pagos,setPagos]=useState(null);
  const [alias,setAlias]=useState("");
  const [acct,setAcct]=useState(null);
  useEffect(()=>{(async()=>{
    setPagos(await sGet("pagos:mine",false)||[]);
    setAcct(await enforceAccount());
    const am=(await getActivePayMethods()).map(m=>`${m.icon?m.icon+" ":""}${m.name}`);
    const names=am.length?am:["Otro"]; setMetodos(names); setMetodo(names[0]);
    let a=await sGet("chat:alias",false);
    if(!a){ a="Cliente-"+Math.random().toString(36).slice(2,7).toUpperCase(); await sSet("chat:alias",a,false); }
    setAlias(a);
  })();},[]);
  const borrar=async()=>{ if(!window.confirm("¿Eliminar tu cuenta y tus datos en este dispositivo? No se puede deshacer.")) return; setAcct(await deleteAccount()); setPagos([]); setMsg("Cuenta eliminada. Confirma un pago para reactivarla."); };
  const enviar=async()=>{
    if(busy) return;
    if(!recibo.trim() && !tx.trim()){ setMsg("Pega el recibo o escribe el número de transacción."); return; }
    setBusy(true); setMsg("");
    const ref="HB-PAY-"+Math.random().toString(36).slice(2,7).toUpperCase();
    const texto=`Método: ${metodo}\nMonto: ${monto||"—"}\nTransacción: ${tx||"—"}`+(recibo.trim()?`\nRecibo:\n${recibo.trim()}`:"");
    let delivered=false;
    try{
      const r=await fetch("/api/ticket",{method:"POST",headers:{"Content-Type":"application/json"},
        body:JSON.stringify({tipo:"pago",alias,contacto:contacto.trim(),texto,ref})});
      const d=await r.json().catch(()=>({})); delivered=!!d.delivered;
    }catch{}
    const rec={ref,metodo,monto:monto.trim(),tx:tx.trim(),recibo:recibo.trim(),contacto:contacto.trim(),ts:Date.now(),delivered,status:"en revisión"};
    const next=[rec,...(pagos||[])].slice(0,50);
    setPagos(next); await sSet("pagos:mine",next,false);
    // CRM: record to the shared store so the owner's Finanzas/CRM sees it.
    try{ const crm=await sGet("crm:pagos",true)||[]; await sSet("crm:pagos",[{ref,alias,metodo,monto:monto.trim(),tx:tx.trim(),contacto:contacto.trim(),ts:Date.now()},...crm].slice(0,1000),true); }catch{}
    const a=await activateAccount(ref); setAcct(a);
    setMsg(`✅ Pago enviado · ${ref}. Hermes ACTIVÓ tu cuenta hasta ${new Date(a.expires).toLocaleDateString("es-ES")}.`+(delivered?" El dueño ya lo recibió.":" El dueño lo confirmará pronto."));
    setMonto(""); setTx(""); setRecibo("");
    setBusy(false);
  };
  const copiar=async(p)=>{
    const txt=`HERMES EL BOLITERO · Confirmación de pago ${p.ref}\n${p.metodo} · ${p.monto||"—"} · Tx ${p.tx||"—"}\nEstado: ${p.status}`;
    try{ if(navigator.share) await navigator.share({text:txt}); else { await navigator.clipboard.writeText(txt); setMsg("Confirmación copiada ✔"); } }catch{}
  };
  const active=accountActive(acct);
  return (
    <div className="fade-in">
      {acct&&(
        <div className="card" style={{marginBottom:12,textAlign:"center",border:active?"1.5px solid #38D6BA":`1px solid ${T.line}`,background:active?"linear-gradient(135deg,rgba(56,214,186,.14),rgba(19,34,43,.9))":T.surface}}>
          <div className="disp" style={{fontWeight:800,color:active?T.teal:T.dim}}>{active?"✅ Cuenta ACTIVA":"🔒 Cuenta inactiva"}</div>
          <div style={{fontSize:12,color:T.dim,marginTop:4}}>{active?`Hermes activó tu cuenta · válida hasta ${new Date(acct.expires).toLocaleDateString("es-ES")}`:"Confirma un pago y Hermes activa tu cuenta al instante."}</div>
        </div>
      )}
      <Section title="Confirmar pago 💳" sub="Paga por tu canal y pega aquí el recibo — el dueño lo confirma">
        <div className="card" style={{display:"grid",gap:10}}>
          <div style={{display:"flex",flexWrap:"wrap",gap:8}}>{metodos.map(m=><button key={m} className={"chip"+(metodo===m?" on":"")} onClick={()=>setMetodo(m)}>{m}</button>)}</div>
          <div style={{display:"flex",gap:8}}>
            <div style={{flex:1}}><Label>Monto</Label>
              <input className="input mono" inputMode="decimal" placeholder="$0.00" value={monto} onChange={e=>setMonto(e.target.value)}/></div>
            <div style={{flex:2}}><Label>N.º de transacción / referencia</Label>
              <input className="input mono" placeholder="ID del pago" value={tx} onChange={e=>setTx(e.target.value)}/></div>
          </div>
          <div><Label>Recibo / comprobante</Label>
            <textarea className="input" rows={4} placeholder="Pega aquí el recibo del pago (PayPal, Cash App, Zelle, captura de la transacción…)" value={recibo} onChange={e=>setRecibo(e.target.value)} style={{resize:"none"}}/></div>
          <input className="input" placeholder="Tu contacto (teléfono, Telegram, email) — opcional" value={contacto} onChange={e=>setContacto(e.target.value)}/>
          <button className="btn primary" onClick={enviar} disabled={busy}>{busy?"Enviando…":"💳 Enviar pago para confirmación"}</button>
          {msg&&<div style={{fontSize:12,color:msg.includes("✅")||msg.includes("✔")?T.teal:T.coral}}>{msg}</div>}
        </div>
      </Section>
      <Section title="Mis pagos" right={<span style={{fontSize:11,color:T.dim}}>{(pagos||[]).length}</span>}>
        <div style={{display:"grid",gap:10}}>
          {pagos&&pagos.length===0&&<div className="card" style={{color:T.dim,fontSize:13,textAlign:"center"}}>Aún no has enviado ningún pago.</div>}
          {(pagos||[]).map(p=>(
            <div key={p.ref} className="card" style={{padding:"12px 14px"}}>
              <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:6}}>
                <span className="mono" style={{fontSize:12,fontWeight:700,color:T.mango}}>{p.ref}</span>
                <span className="chip" style={{cursor:"default"}}>{p.delivered?"📨 en revisión":"⏳ guardado"}</span>
              </div>
              <div style={{fontSize:13}}>{p.metodo}{p.monto?` · ${p.monto}`:""}{p.tx?` · Tx ${p.tx}`:""}</div>
              {p.recibo&&<div style={{fontSize:12,color:T.dim,whiteSpace:"pre-wrap",marginTop:6,maxHeight:80,overflow:"hidden"}}>{p.recibo}</div>}
              <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginTop:8}}>
                <span style={{fontSize:10,color:T.dim}}>{new Date(p.ts).toLocaleString("es-ES",{day:"numeric",month:"short",hour:"2-digit",minute:"2-digit"})}</span>
                <button className="chip" onClick={()=>copiar(p)}>📤 Copiar confirmación</button>
              </div>
            </div>
          ))}
        </div>
      </Section>
      <div style={{fontSize:11,color:T.dim,lineHeight:1.6,marginBottom:10}}>
