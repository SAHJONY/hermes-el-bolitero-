/* Claves compartidas:
   members:codes  -> {"CODE":{label,active,created,uses}}
   admin:pin      -> pin del dueño (se crea la primera vez, solo tú)            */
async function sGet(key,shared){ try{const r=await window.storage.get(key,shared); return r?JSON.parse(r.value):null;}catch{return null;} }
async function sSet(key,val,shared){ try{await window.storage.set(key,JSON.stringify(val),shared); return true;}catch{return false;} }

function Gate({onEnter}){
  const [stage,setStage]=useState("loading"); // loading | code | newadmin | adminpin | planes
  const [val,setVal]=useState("");
  const [err,setErr]=useState("");
  const [payLinks,setPayLinks]=useState(null);
  useEffect(()=>{(async()=>{
    const sess=await sGet("session",false);
    if(sess?.ok){ onEnter(sess.admin===true); return; }
    setPayLinks(await sGet("pay:links",true));
    const pin=await sGet("admin:pin",true);
    setStage(pin?"code":"newadmin");
  })();},[]);
  const enterMember=async()=>{
    setErr("");
    const codes=await sGet("members:codes",true)||{};
    const c=codes[val.trim().toUpperCase()];
    if(c&&c.active){
      c.uses=(c.uses||0)+1; await sSet("members:codes",codes,true);
      await sSet("session",{ok:true,admin:false,code:val.trim().toUpperCase()},false);
      onEnter(false);
    } else setErr("Código inválido o desactivado. Pídele acceso al administrador.");
  };
  const enterAdmin=async()=>{
    setErr("");
    const pin=await sGet("admin:pin",true);
    if(pin&&pin.pin===val.trim()){ await sSet("session",{ok:true,admin:true},false); onEnter(true); }
    else setErr("PIN incorrecto.");
  };
  const createAdmin=async()=>{
    setErr("");
    if(val.trim().length<4){ setErr("Usa un PIN de al menos 4 dígitos."); return; }
    const exists=await sGet("admin:pin",true);
    if(exists){ setErr("Ya existe un administrador. Entra con tu PIN."); setStage("adminpin"); setVal(""); return; }
    await sSet("admin:pin",{pin:val.trim(),created:Date.now()},true);
    await sSet("members:codes",{},true);
    await sSet("session",{ok:true,admin:true},false);
    onEnter(true);
  };
  if(stage==="loading") return <GateShell><div style={{color:T.dim}}>Cargando…</div></GateShell>;
  if(stage==="planes") return <Planes payLinks={payLinks} onBack={()=>setStage("code")}/>;
  return (
    <GateShell>
      {stage==="newadmin"&&(<>
        <div className="disp" style={{fontWeight:800,fontSize:20,marginBottom:6}}>Bienvenido, dueño 👑</div>
        <p style={{fontSize:13,color:T.dim,margin:"0 0 14px"}}>Eres el primero en entrar. Crea tu PIN de administrador — solo tú controlarás quién accede.</p>
        <input className="input" type="password" inputMode="numeric" placeholder="Crea tu PIN de admin" value={val} onChange={e=>setVal(e.target.value)} onKeyDown={e=>e.key==="Enter"&&createAdmin()}/>
        <button className="btn primary" style={{width:"100%",marginTop:10}} onClick={createAdmin}>Crear y entrar</button>
      </>)}
      {stage==="code"&&(<>
        <div className="disp" style={{fontWeight:800,fontSize:20,marginBottom:6}}>Solo miembros 🎟️</div>
        <p style={{fontSize:13,color:T.dim,margin:"0 0 14px"}}>Escribe tu código de acceso.</p>
        <input className="input" placeholder="CÓDIGO" value={val} onChange={e=>setVal(e.target.value.toUpperCase())} onKeyDown={e=>e.key==="Enter"&&enterMember()} style={{textTransform:"uppercase",letterSpacing:"0.15em",textAlign:"center",fontFamily:"IBM Plex Mono"}}/>
        <button className="btn primary" style={{width:"100%",marginTop:10}} onClick={enterMember}>Entrar</button>
        <button className="btn ghost" style={{width:"100%",marginTop:10}} onClick={()=>setStage("planes")}>Ver planes · desde $19.99/mes</button>
        <button className="chip" style={{marginTop:14}} onClick={()=>{setStage("adminpin");setVal("");setErr("");}}>👑 Soy el administrador</button>
      </>)}
      {stage==="adminpin"&&(<>
        <div className="disp" style={{fontWeight:800,fontSize:20,marginBottom:6}}>Acceso admin 👑</div>
        <input className="input" type="password" inputMode="numeric" placeholder="PIN de administrador" value={val} onChange={e=>setVal(e.target.value)} onKeyDown={e=>e.key==="Enter"&&enterAdmin()}/>
        <button className="btn primary" style={{width:"100%",marginTop:10}} onClick={enterAdmin}>Entrar como admin</button>
        <button className="chip" style={{marginTop:14}} onClick={()=>{setStage("code");setVal("");setErr("");}}>Soy miembro</button>
      </>)}
      {err&&<div style={{color:T.coral,fontSize:12,marginTop:12}}>{err}</div>}
    </GateShell>
  );
}

    const saved=await sGet("env",false);
    const legacy=await sGet("cfg",false);
    const merged={...ENV_DEFAULTS,
      ...(legacy?{ENGINE_MODEL:legacy.hermesModel||ENV_DEFAULTS.ENGINE_MODEL,ENGINE_API_KEY:legacy.hermesKey||"",VOICE_API_KEY:legacy.blandKey||"",VOICE_FROM_NUMBER:legacy.blandFrom||"",MAIL_API_KEY:legacy.emailKey||"",MAIL_FROM:legacy.emailFrom||ENV_DEFAULTS.MAIL_FROM}:{}),

              <button className="chip" style={{color:T.coral}} onClick={()=>remove(c)}>✕</button>
            </div>
          ))}
          {codes&&Object.keys(codes).length===0&&<div className="card" style={{color:T.dim,fontSize:13,textAlign:"center"}}>Aún no hay códigos. Crea el primero y compártelo junto al enlace de la app.</div>}
        </div>
      </Section>
      <div style={{fontSize:11,color:T.dim,lineHeight:1.6}}>
        La app funciona en cualquier dispositivo — iPhone, Android, tablet o computadora — con el mismo enlace publicado + código.
      </div>
      </>)}
      {view==="seguridad"&&(<>
      <Section title="Seguridad" sub="Cambiar PIN de administrador">
        <div className="card" style={{display:"flex",gap:8}}>
          <input className="input" type="password" placeholder="Nuevo PIN" value={pinNew} onChange={e=>setPinNew(e.target.value)}/>
          <button className="btn ghost" onClick={changePin}>Cambiar</button>
        </div>
      </Section>
      {saved&&<div style={{color:T.teal,fontSize:12,textAlign:"center"}}>{saved}</div>}
      <div style={{fontSize:11,color:T.dim,marginTop:10,lineHeight:1.6}}>
        Para compartir: publica esta app con el botón <b>Compartir/Publicar</b> del artefacto y entrega el enlace + un código.
        Esta puerta es un control ligero de comunidad; para producción real usa el backend incluido con autenticación de verdad.
      </div>
      </>)}
    </div>
  );
}

/* ============================ APP (con puerta de miembros) ============================ */
function App(){
  const [auth,setAuth]=useState(null); // null=gate, {admin:bool}
  const lock=async()=>{ try{await window.storage.delete("session");}catch{} setAuth(null); };
  if(auth===null) return <Gate onEnter={(admin)=>setAuth({admin})}/>;
  return <MainApp admin={auth.admin} onLock={lock}/>;
}

ReactDOM.createRoot(document.getElementById('root')).render(React.createElement(App));
