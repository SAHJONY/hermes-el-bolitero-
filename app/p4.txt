async function sGet(key,shared){ try{const r=await window.storage.get(key,shared); return r?JSON.parse(r.value):null;}catch{return null;} }
async function sSet(key,val,shared){ try{await window.storage.set(key,JSON.stringify(val),shared); return true;}catch{return false;} }
async function sDel(key,shared){ try{await window.storage["delete"](key,!!shared); return true;}catch{return false;} }

/* ---------- Cuenta del cliente: Hermes activa con el pago y borra al vencer ---------- */
const ACCOUNT_DAYS=30;
async function getAccount(){ return (await sGet("account",false))||{status:"inactivo",expires:0}; }
function accountActive(a){ return !!(a && a.status==="activo" && a.expires && Date.now() < a.expires); }
async function activateAccount(ref){
  const a={ status:"activo", since:Date.now(), expires:Date.now()+ACCOUNT_DAYS*86400000, ref:ref||null };
  await sSet("account",a,false); return a;
}
// Autonomous: deactivate + wipe private data when the membership lapses.
async function enforceAccount(){
  const a=await getAccount();
  if(a.status==="activo" && a.expires && Date.now() >= a.expires){
    await sSet("account",{status:"vencida",expires:a.expires,ref:a.ref,deletedAt:Date.now()},false);
    await sDel("chat:hist",false); await sDel("recibos:mine",false); await sDel("tickets:mine",false);
    return await getAccount();
  }
  return a;
}
async function deleteAccount(){
  await sSet("account",{status:"inactivo",expires:0,deletedAt:Date.now()},false);
  await sDel("chat:hist",false); await sDel("recibos:mine",false); await sDel("tickets:mine",false); await sDel("pagos:mine",false);
  return await getAccount();
}

/* ---------- Monedas virtuales (entretenimiento) + saldo en tiempo real ----------
   Saldo de juego en MONEDAS virtuales (no dinero real). Se guarda por cuenta y
   se refleja al instante; se replica al store compartido (bal:<alias>) para el
   seguimiento global del saldo de la app vía /api/balance.
   La CONFIGURACIÓN (saldo inicial, bono, multiplicadores, on/off) la controla el
   dueño desde el panel 👑 → 🪙 Monedas y se guarda en el store compartido
   "coins:config" para que aplique a TODOS los usuarios.                          */
const COINS_DEFAULTS={ enabled:true, start:1000, daily:100, payFijo:75, payCorrido:25 };
let _coinsCfg=null;
async function getCoinsConfig(){
  if(_coinsCfg) return _coinsCfg;
  const c=await sGet("coins:config",true);
  _coinsCfg={...COINS_DEFAULTS, ...(c||{})};
  return _coinsCfg;
}
async function setCoinsConfig(patch){
  const cur=await getCoinsConfig();
  const next={...cur,...patch};
  _coinsCfg=next;
  await sSet("coins:config",next,true);
  return next;
}
async function coinsAlias(){ let a=await sGet("chat:alias",false); if(!a){ a="Cliente-"+Math.random().toString(36).slice(2,7).toUpperCase(); await sSet("chat:alias",a,false);} return a; }
async function getCoins(){ const cfg=await getCoinsConfig(); const c=await sGet("coins",false); return (c&&typeof c.bal==="number")?c:{bal:cfg.start,lastBonus:""}; }
async function setCoins(c){ await sSet("coins",c,false); try{ const a=await coinsAlias(); await sSet("bal:"+a,{bal:c.bal,ts:Date.now()},true); }catch(e){} return c; }
async function claimDailyBonus(){ const cfg=await getCoinsConfig(); if(!cfg.enabled||cfg.daily<=0) return {claimed:false,bal:(await getCoins()).bal}; const c=await getCoins(); const today=new Date().toISOString().slice(0,10); if(c.lastBonus!==today){ c.bal+=cfg.daily; c.lastBonus=today; await setCoins(c); return {claimed:true,bal:c.bal}; } return {claimed:false,bal:c.bal}; }
async function spendCoins(n){ const c=await getCoins(); if(c.bal<n) return {ok:false,bal:c.bal}; c.bal-=n; await setCoins(c); return {ok:true,bal:c.bal}; }
async function addCoins(n){ const c=await getCoins(); c.bal=Math.max(0,c.bal+Math.round(n)); await setCoins(c); return c.bal; }
async function resetMyCoins(to){ const cfg=await getCoinsConfig(); const c={bal:(typeof to==="number"?to:cfg.start),lastBonus:""}; await setCoins(c); return c.bal; }

  })();},[]);

function App(){
  const [auth,setAuth]=useState(null); // null=gate, {admin:bool}
  const lock=async()=>{ try{await window.storage.delete("session");}catch{} setAuth(null); };
  if(auth===null) return <Gate onEnter={(admin)=>setAuth({admin})}/>;
  return <MainApp admin={auth.admin} onLock={lock}/>;
}
