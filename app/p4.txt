/* Claves compartidas:
   members:codes  -> {"CODE":{label,active,created,uses}}
   admin:pin      -> pin del dueño (se crea la primera vez, solo tú)            */
async function sGet(key,shared){ try{const r=await window.storage.get(key,shared); return r?JSON.parse(r.value):null;}catch{return null;} }
async function sSet(key,val,shared){ try{await window.storage.set(key,JSON.stringify(val),shared); return true;}catch{return false;} }
async function sDel(key,shared){ try{await window.storage["delete"](key,!!shared); return true;}catch{return false;} }
// Llama una operación del cerebro/motor en /api/db (game.*, ticket.*). Devuelve
// null si el almacén no está conectado (el llamador cae a la lógica local).
async function dbOp(op,payload){ try{ const r=await fetch("/api/db",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify(Object.assign({op:op},payload||{}))}); if(!r.ok) return null; const d=await r.json(); return (d&&d.configured===false)?null:d; }catch(e){ return null; } }

/* ---------- Cuenta del cliente: Hermes activa con el pago y borra al vencer ---------- */
const ACCOUNT_DAYS=30;
async function getAccount(){ return (await sGet("account",false))||{status:"inactivo",expires:0}; }
function accountActive(a){ return !!(a && a.status==="activo" && a.expires && Date.now() < a.expires); }
async function activateAccount(ref){
  const a={ status:"activo", since:Date.now(), expires:Date.now()+ACCOUNT_DAYS*86400000, ref:ref||null };
  await sSet("account",a,false); return a;
}
// Autonomous: deactivate + wipe private data when the membership lapses.
async function enforceAccount(){
  const a=await getAccount();
  if(a.status==="activo" && a.expires && Date.now() >= a.expires){
    await sSet("account",{status:"vencida",expires:a.expires,ref:a.ref,deletedAt:Date.now()},false);
    await sDel("chat:hist",false); await sDel("recibos:mine",false); await sDel("tickets:mine",false);
    return await getAccount();
  }
  return a;
}
async function deleteAccount(){
  await sSet("account",{status:"inactivo",expires:0,deletedAt:Date.now()},false);
  await sDel("chat:hist",false); await sDel("recibos:mine",false); await sDel("tickets:mine",false); await sDel("pagos:mine",false);
  return await getAccount();
}

/* ---------- Monedas virtuales (entretenimiento) + saldo en tiempo real ----------
   Saldo de juego en MONEDAS virtuales (no dinero real). Se guarda por cuenta y
   se refleja al instante; se replica al store compartido (bal:<alias>) para el
   seguimiento global del saldo de la app vía /api/db (op:"balance").
   La CONFIGURACIÓN (saldo inicial, bono, multiplicadores, on/off) la controla el
   dueño desde el panel 👑 → 🪙 Monedas y se guarda en el store compartido
   "coins:config" para que aplique a TODOS los usuarios.                          */
const COINS_DEFAULTS={ enabled:true, start:1000, daily:100, payFijo:56, payCorrido:25, payParle:1300, payCandado:10000, payPlay4:5000, payQ1:60, payQ2:20, payQ3:10, payCash3:500, payBox3:160, payBox4:1200, payPair:70, payCentena:600 };
let _coinsCfg=null;
async function getCoinsConfig(){
  if(_coinsCfg) return _coinsCfg;
  const c=await sGet("coins:config",true);
  _coinsCfg={...COINS_DEFAULTS, ...(c||{})};
  return _coinsCfg;
}
async function setCoinsConfig(patch){
  const cur=await getCoinsConfig();
  const next={...cur,...patch};
  _coinsCfg=next;
  await sSet("coins:config",next,true);
  return next;
}
async function coinsAlias(){ let a=await sGet("chat:alias",false); if(!a){ a="Cliente-"+Math.random().toString(36).slice(2,7).toUpperCase(); await sSet("chat:alias",a,false);} return a; }
// Saldo SERVER-AUTHORITATIVE: vive en el almacén compartido (wallet:<alias> vía Supabase),
// así es global entre dispositivos y el Banco Central lo ve/controla. Cae a localStorage si
// el almacén no está conectado (sin romper). Migra una sola vez el viejo saldo local.
async function getCoins(){
  const cfg=await getCoinsConfig(); const a=await coinsAlias();
  let c=await sGet("wallet:"+a,true);
  if(!c){ const local=await sGet("coins",false); if(local&&typeof local.bal==="number") c=local; }
  return (c&&typeof c.bal==="number")?{streak:0,bestStreak:0,xp:0,...c}:{bal:cfg.start,lastBonus:"",streak:0,bestStreak:0,xp:0};
}
async function setCoins(c){
  const a=await coinsAlias();
  await sSet("wallet:"+a,c,true);                              // saldo global (servidor)
  try{ await sSet("coins",c,false); }catch(e){}                // espejo local (rápido/offline)
  try{ await sSet("bal:"+a,{bal:c.bal,ts:Date.now()},true); }catch(e){} // índice de saldos para el panel
  return c;
}
const STREAK_MULT=[0.5,0.75,1,1.5,2,3,10]; // día 1..7+ : el día 7 es jackpot ×10
async function claimDailyBonus(){
  const cfg=await getCoinsConfig();
  if(!cfg.enabled) return {claimed:false,bal:(await getCoins()).bal};
  const c=await getCoins();
  const today=new Date().toISOString().slice(0,10);
  if(c.lastBonus===today) return {claimed:false,bal:c.bal,streak:c.streak||0};
  const ayer=new Date(Date.now()-86400000).toISOString().slice(0,10);
  c.streak = (c.lastBonus===ayer) ? (c.streak||0)+1 : 1;
  c.bestStreak=Math.max(c.bestStreak||0,c.streak);
  const idx=Math.min(c.streak-1, STREAK_MULT.length-1);
  const amount=Math.max(1,Math.round((cfg.daily||100)*STREAK_MULT[idx]));
  c.bal+=amount; c.lastBonus=today; c.xp=(c.xp||0); 
  await setCoins(c);
  return {claimed:true,bal:c.bal,amount,streak:c.streak,jackpot:c.streak>=7};
}
async function spendCoins(n){ const c=await getCoins(); if(c.bal<n) return {ok:false,bal:c.bal}; c.bal-=n; await setCoins(c); return {ok:true,bal:c.bal}; }
async function addCoins(n){ const c=await getCoins(); n=Math.round(n); c.bal=Math.max(0,c.bal+n); if(n>0) c.xp=(c.xp||0)+n; await setCoins(c); return c.bal; }
async function resetMyCoins(to){ const cfg=await getCoinsConfig(); const c={bal:(typeof to==="number"?to:cfg.start),lastBonus:""}; await setCoins(c); return c.bal; }

/* ---------- NIVELES / RANGOS (prestigio por monedas ganadas) ---------- */
const RANKS=[
  {min:0,name:"Novato",icon:"🥉"},
  {min:1000,name:"Bolitero",icon:"🥈"},
  {min:10000,name:"Maestro",icon:"🥇"},
  {min:50000,name:"Leyenda",icon:"👑"},
  {min:200000,name:"Mito",icon:"⚡"},
];
function rankOf(xp){
  xp=xp||0; let r=RANKS[0];
  for(const x of RANKS){ if(xp>=x.min) r=x; }
  const i=RANKS.indexOf(r); const next=RANKS[i+1]||null;
  return {...r, next, progress: next? Math.max(0,Math.min(1,(xp-r.min)/(next.min-r.min))) : 1};
}

/* ---------- MISIONES DIARIAS (local por dispositivo, se reinician cada día) ---------- */
const MISSIONS=[
  {id:"play3", txt:"Juega 3 tickets", goal:3, reward:150, icon:"🎟️"},
  {id:"check", txt:"Comprueba tus tickets", goal:1, reward:50, icon:"🔎"},
  {id:"win1", txt:"Acierta una jugada", goal:1, reward:300, icon:"🏆"},
];
async function getMissions(){
  const today=new Date().toISOString().slice(0,10);
  let m=await sGet("missions",false);
  if(!m||m.date!==today){ m={date:today,prog:{},claimed:{}}; await sSet("missions",m,false); }
  return m;
}
async function bumpMission(id,by){
  const m=await getMissions(); m.prog[id]=(m.prog[id]||0)+(by||1); await sSet("missions",m,false); return m;
}
async function claimMission(id){
  const m=await getMissions(); const def=MISSIONS.find(x=>x.id===id);
  if(!def||m.claimed[id]||(m.prog[id]||0)<def.goal) return {ok:false};
  m.claimed[id]=true; await sSet("missions",m,false); await addCoins(def.reward);
  return {ok:true,reward:def.reward};
}

/* ---------- RANKING SEMANAL DE JUGADORES (compartido) ---------- */
function weekKey(){
  const d=new Date(); const onejan=new Date(d.getFullYear(),0,1);
  const week=Math.ceil((((d-onejan)/86400000)+onejan.getDay()+1)/7);
  return d.getFullYear()+"-W"+week;
}
async function pushScore(name,xp){
  const wk=weekKey(); let lb=await sGet("leaderboard",true);
  if(!lb||lb.week!==wk) lb={week:wk,players:{}};
  const a=await coinsAlias();
  const prev=(lb.players[a]&&lb.players[a].score)||0;
  lb.players[a]={name:name||a,score:Math.max(prev, Math.round(xp||0))};
  await sSet("leaderboard",lb,true); return lb;
}
async function getLeaderboard(){
  const wk=weekKey(); const lb=await sGet("leaderboard",true);
  if(!lb||lb.week!==wk) return {week:wk,players:{}};
  return lb;
}

/* ---------- MINI-JUEGOS DIARIOS (Hot Ball + Charada Challenge) ----------
   Una jugada gratis por día de cada mini-juego, recompensa en monedas virtuales.
   Estado local por dispositivo: minigames -> { date, hotball:{played,won}, charada:{played,won} } */
const MINIGAME_REWARD={ hotball:200, charada:150 };
async function getMiniState(){
  const today=new Date().toISOString().slice(0,10);
  let m=await sGet("minigames",false);
  if(!m||m.date!==today){ m={date:today, hotball:{played:false}, charada:{played:false}}; await sSet("minigames",m,false); }
  return m;
}
async function setMiniState(m){ await sSet("minigames",m,false); return m; }
// "Hot Ball": el jugador adivina si un número estará entre los 5 más calientes del tablero.
async function playHotBall(lotId, guess){
  const m=await getMiniState(); if(m.hotball.played) return {already:true};
  const st=buildStats(lotId); const hot=st.hot.slice(0,5).map(x=>x.n);
  const won=hot.includes(parseInt(guess,10));
  m.hotball={played:true, won, guess:parseInt(guess,10), hot};
  await setMiniState(m);
  if(won){ await addCoins(MINIGAME_REWARD.hotball); try{ await bumpMission("win1"); }catch{} }
  return {won, hot, reward:won?MINIGAME_REWARD.hotball:0};
}
// "Charada Challenge": dan un símbolo, el jugador elige el número correcto entre opciones.
async function playCharada(correctN, pickN){
  const m=await getMiniState(); if(m.charada.played) return {already:true};
  const won=parseInt(correctN,10)===parseInt(pickN,10);
  m.charada={played:true, won};
  await setMiniState(m);
  if(won){ await addCoins(MINIGAME_REWARD.charada); }
  return {won, reward:won?MINIGAME_REWARD.charada:0};
}

/* ---------- SUCURSALES estilo Monopoly (Banco Central → Sucursales) ----------
   100% monedas virtuales. El Banco Central (dueño) crea sucursales; cada sucursal
   reparte monedas a SUS jugadores y por cada jugada virtual una comisión % fluye
   al Banco Central. Todo compartido vía /api/db para verse entre dispositivos.
   Claves compartidas:
     bank:config         -> { commissionPct }
     bank:central        -> { coins }                 (bolsa del Banco Central)
     branches            -> { "CODE": {name,active,created,coins,plays,commission,wagered} }
   Local (por dispositivo): branch:me -> code de la sucursal en la que juega este device */
const BANK_DEFAULTS={ commissionPct:10 };
async function getBankConfig(){ const c=await sGet("bank:config",true); return {...BANK_DEFAULTS, ...(c||{})}; }
async function setBankConfig(patch){ const cur=await getBankConfig(); const next={...cur,...patch}; await sSet("bank:config",next,true); return next; }
async function getBranches(){ return (await sGet("branches",true))||{}; }
async function getCentral(){ return (await sGet("bank:central",true))||{coins:0}; }
async function newBranch(name){
  const br=await getBranches();
  const code=Array.from({length:5},()=> "ABCDEFGHJKMNPQRSTUVWXYZ23456789"[Math.floor(Math.random()*31)]).join("");
  br[code]={ name:name||"Sucursal", active:true, created:Date.now(), coins:0, plays:0, commission:0, wagered:0 };
  await sSet("branches",br,true); return code;
}
async function setBranchActive(code,on){ const br=await getBranches(); if(br[code]){ br[code].active=on; await sSet("branches",br,true);} return br; }
async function removeBranch(code){ const br=await getBranches(); delete br[code]; await sSet("branches",br,true); return br; }
async function myBranch(){ return await sGet("branch:me",false); }       // {code,name}
async function joinBranch(code){
  const br=await getBranches(); const b=br[String(code).toUpperCase()];
  if(!b||!b.active) return null;
  const me={code:String(code).toUpperCase(),name:b.name};
  await sSet("branch:me",me,false); return me;
}
// Registrar una jugada virtual: cobra comisión a la sucursal y la pasa al Banco Central.
async function recordWager(amount){
  amount=Math.max(0,Math.round(amount||0)); if(!amount) return;
  const cfg=await getBankConfig(); const pct=Math.max(0,Math.min(100,cfg.commissionPct||0));
  const fee=Math.round(amount*pct/100);
  const me=await myBranch();
  if(me){ const br=await getBranches(); const b=br[me.code]; if(b){ b.plays=(b.plays||0)+1; b.wagered=(b.wagered||0)+amount; b.commission=(b.commission||0)+fee; await sSet("branches",br,true); } }
  // TODO cliente es del Banco Central: cobra comisión de cada jugada (con o sin sucursal) y la registra.
  const c=await getCentral(); c.plays=(c.plays||0)+1; c.wagered=(c.wagered||0)+amount; c.coins=(c.coins||0)+fee; await sSet("bank:central",c,true);
}
// Banco Central GOD-MODE (dueño): acuña/ajusta la bolsa central y da/quita
// monedas a cualquier sucursal — sin límite. 100% monedas virtuales.
async function setCentralCoins(coins){ const c=await getCentral(); c.coins=Math.max(0,Math.round(Number(coins)||0)); await sSet("bank:central",c,true); return c; }
async function addCentralCoins(delta){ const c=await getCentral(); c.coins=Math.max(0,Math.round((c.coins||0)+(Number(delta)||0))); await sSet("bank:central",c,true); return c; }
async function giveBranchCoins(code,delta){ const br=await getBranches(); const b=br[code]; if(b){ b.coins=Math.max(0,Math.round((b.coins||0)+(Number(delta)||0))); await sSet("branches",br,true);} return br; }
// Fijar tu propio saldo (dueño) a cualquier número.
async function setMyBalance(to){ const c=await getCoins(); c.bal=Math.max(0,Math.round(Number(to)||0)); await setCoins(c); return c.bal; }
// Dar/quitar monedas a un miembro por su código. Se acumula como total y el
// dispositivo del miembro lo aplica a su saldo local cuando abre la app.
async function grantToMember(code,delta){ const k="grant:"+String(code||"").toUpperCase(); const g=(await sGet(k,true))||{total:0}; g.total=Math.round((g.total||0)+(Number(delta)||0)); g.ts=Date.now(); await sSet(k,g,true); return g; }
async function applyGrants(){
  try{
    const sess=await sGet("session",false); const code=sess&&sess.code; if(!code) return 0;
    const g=await sGet("grant:"+code,true); if(!g) return 0;
    const claimed=Number(await sGet("grant:claimed",false))||0;
    const delta=(Number(g.total)||0)-claimed; if(!delta) return 0;
    const c=await getCoins(); c.bal=Math.max(0,Math.round((c.bal||0)+delta)); await setCoins(c);
    await sSet("grant:claimed",g.total,false); return delta;
  }catch(e){ return 0; }
}
// Resetear toda la economía compartida: bolsa central, sucursales y ranking.
async function resetEconomy(){
  await sSet("bank:central",{coins:0},true);
  const br=await getBranches(); Object.keys(br).forEach(k=>{ br[k].coins=0; br[k].commission=0; br[k].plays=0; br[k].wagered=0; }); await sSet("branches",br,true);
  try{ await sSet("leaderboard",{week:"",players:{}},true); }catch(e){}
  return true;
}

  return all.filter(m=> am ? !!am[m.id] : PAY_DEFAULT_ACTIVE.includes(m.id));
}

/* ---------- Seguridad del acceso (Opción A): PIN propio por cliente, código
   amarrado al dispositivo, PIN de admin cifrado (hash) y límite de intentos.
   Privado: NO se piden email ni teléfono. ---------- */
async function hbHash(s){
  try{ const b=await crypto.subtle.digest("SHA-256", new TextEncoder().encode("hb1:"+String(s))); return [...new Uint8Array(b)].map(x=>x.toString(16).padStart(2,"0")).join(""); }
  catch(e){ let h=2166136261>>>0; const t="hb1:"+String(s); for(let i=0;i<t.length;i++){ h^=t.charCodeAt(i); h=Math.imul(h,16777619); } return "f"+(h>>>0).toString(16); }
}
async function getDeviceId(){ let d=await sGet("device:id",false); if(!d){ d="dev-"+Math.random().toString(36).slice(2,10)+Date.now().toString(36); await sSet("device:id",d,false);} return d; }
async function authLockedSecs(){ const l=await sGet("auth:lock",false); return (l&&l.until&&Date.now()<l.until)?Math.ceil((l.until-Date.now())/1000):0; }
async function authFail(){ const l=(await sGet("auth:lock",false))||{fails:0}; l.fails=(l.fails||0)+1; if(l.fails>=5){ l.until=Date.now()+60000; l.fails=0; } await sSet("auth:lock",l,false); }
async function authReset(){ await sDel("auth:lock",false); }

  const toggle=async(c)=>{ const next={...codes,[c]:{...codes[c],active:!codes[c].active}}; await sSet("members:codes",next,true); setCodes(next); };
  const remove=async(c)=>{ const next={...codes}; delete next[c]; await sSet("members:codes",next,true); setCodes(next); };
  const unbind=async(c)=>{ const next={...codes,[c]:{...codes[c],boundTo:""}}; await sSet("members:codes",next,true); setCodes(next); setSaved("Código "+c+" desvinculado — ya puede usarse en otro dispositivo."); };
  const changePin=async()=>{ if(pinNew.trim().length<4) return; await sSet("admin:pin",{pin:pinNew.trim(),created:Date.now()},true); setPinNew(""); setSaved("PIN actualizado."); };
  return (
    <div className="fade-in">
      <div style={{display:"flex",alignItems:"center",gap:10,marginBottom:12}}>
        <span style={{fontSize:30,filter:"drop-shadow(0 0 10px rgba(255,184,56,.7)) saturate(1.5)"}}>👑</span>
        <div>
          <div className="disp" style={{fontWeight:800,fontSize:19,background:"linear-gradient(135deg,#FFE9A8,#FFB838 55%,#E08C00)",WebkitBackgroundClip:"text",backgroundClip:"text",color:"transparent"}}>Panel del Dueño</div>
          <div style={{fontSize:11,color:T.dim}}>Solo tú ves esto, corona en mano.</div>
        </div>
      </div>
      <div className="scroll-x" style={{marginBottom:14}}>
        {[["miembros","🎟️ Miembros"],["finanzas","💰 Finanzas"],["monedas","🪙 Monedas y premios"],["sucursales","🏦 Sucursales"],["alertas","🔔 Alertas"],["anuncios","📢 Anuncios"],["pagos","💳 Pagos"],["env","🔐 ENV · Secretos"],["seguridad","🛡️ Seguridad"]].map(([id,lab])=>(
          <button key={id} className={"chip"+(view===id?" on":"")} onClick={()=>setView(id)}>{lab}</button>
        ))}
      </div>
      {view==="env"&&<EnvPage/>}
      {view==="finanzas"&&<FinanzasPage/>}
      {view==="monedas"&&<MonedasPage/>}
      {view==="sucursales"&&<SucursalesPage/>}
      {view==="pagos"&&<PagosPage/>}
      {view==="anuncios"&&<AnunciosAdmin/>}
      {view==="alertas"&&<AlertasAdmin/>}
      {view==="miembros"&&(<>
      <div className="card" style={{marginBottom:12,padding:14}}>
        <div className="disp" style={{fontWeight:800,fontSize:15,marginBottom:10}}>📊 Comunidad de un vistazo</div>
        <div style={{display:"grid",gridTemplateColumns:"repeat(3,1fr)",gap:8}}>
          {[["👥","Miembros",mstats.total],["✅","Activos",mstats.active],["💎","De pago",mstats.paid],
            ["📲","Con dispositivo",mstats.withDev],["🔓","Entradas",mstats.uses],["🎮","Jugadores",econ?econ.players:"—"]
          ].map(([ic,lab,val],ix)=>(
            <div key={ix} style={{background:T.raise,borderRadius:12,padding:"10px 8px",textAlign:"center"}}>
              <div style={{fontSize:18}}>{ic}</div>
              <div className="disp" style={{fontWeight:800,fontSize:18,color:T.ink}}>{typeof val==="number"?val.toLocaleString():val}</div>
              <div style={{fontSize:10,color:T.dim}}>{lab}</div>
            </div>
          ))}
        </div>
        {econ&&<div style={{marginTop:10,fontSize:12,color:T.dim,textAlign:"center"}}>🪙 {econ.coins.toLocaleString()} monedas en circulación en la comunidad</div>}
        {top&&top.length>0&&(
          <div style={{marginTop:10,borderTop:"1px solid "+T.line,paddingTop:10}}>
            <div style={{fontSize:11,color:T.dim,marginBottom:6}}>🔥 Más activos (por monedas ganadas)</div>
            {top.map((p,ix)=>(
              <div key={ix} style={{display:"flex",justifyContent:"space-between",fontSize:12,padding:"3px 0"}}>
                <span style={{color:T.ink}}>{["🥇","🥈","🥉"][ix]||(ix+1+".")} {p.alias}</span>
                <span style={{color:T.teal}}>{(p.bal||0).toLocaleString()} 🪙 · XP {(p.xp||0).toLocaleString()}</span>
              </div>
            ))}
          </div>
        )}
      </div>
      <Section title="Códigos de miembros" sub="Crea, desactiva o elimina el acceso — solo tú lo controlas">
        <div className="card" style={{marginBottom:10}}>
          <div style={{display:"flex",gap:8}}>
            <input className="input" placeholder="Etiqueta (ej: Tío Pepe)" value={label} onChange={e=>setLabel(e.target.value)}/>
            <button className="btn primary" onClick={newCode}>+ Código</button>
          </div>
        </div>
        <div style={{display:"grid",gap:8}}>
          {codes&&Object.entries(codes).map(([c,info])=>(
            <div key={c} className="card" style={{padding:"12px 14px",display:"flex",alignItems:"center",gap:10}}>
              <span className="mono" style={{fontWeight:700,letterSpacing:"0.12em",color:info.active?T.teal:T.dim}}>{c}</span>
              <span style={{fontSize:12,color:T.dim,flex:1,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{info.nombre||info.label||"Miembro"}{planBadge(info.plan)?" · "+planBadge(info.plan):""} · {info.uses||0} entradas{mstats.refCount[c]?" · 🔗 "+mstats.refCount[c]+" invitados":""}{info.created?" · "+new Date(info.created).toLocaleDateString("es-ES",{day:"numeric",month:"short"}):""}{info.boundTo?" · 🔒":""}</span>
              <button className="chip" style={{color:T.teal}} onClick={async()=>{ const n=parseInt(window.prompt("¿Cuántas monedas dar al miembro "+c+"? (negativo para quitar). Se aplican cuando el miembro abra la app."),10); if(!isNaN(n)){ await grantToMember(c,n); setSaved((n>=0?"+":"")+n.toLocaleString()+" 🪙 → "+c); } }}>💰</button>
              {info.boundTo&&<button className="chip" title="Desvincular dispositivo" onClick={()=>unbind(c)}>🔓</button>}
              <button className="chip" onClick={()=>toggle(c)}>{info.active?"Activo":"Pausado"}</button>
              <button className="chip" style={{color:T.coral}} onClick={()=>remove(c)}>✕</button>
            </div>
          ))}
          {codes&&Object.keys(codes).length===0&&<div className="card" style={{color:T.dim,fontSize:13,textAlign:"center"}}>Aún no hay códigos. Crea el primero y compártelo junto al enlace de la app.</div>}
        </div>
        {saved&&<div style={{color:T.teal,fontSize:12,textAlign:"center",marginTop:8}}>{saved}</div>}
      </Section>
      <div style={{fontSize:11,color:T.dim,lineHeight:1.6}}>
        La app funciona en cualquier dispositivo — iPhone, Android, tablet o computadora — con el mismo enlace publicado + código.
      </div>
      </>)}
      {view==="seguridad"&&(<>
      <Section title="Seguridad" sub="Cambiar PIN de administrador">
        <div className="card" style={{display:"flex",gap:8}}>
          <input className="input" type="password" placeholder="Nuevo PIN" value={pinNew} onChange={e=>setPinNew(e.target.value)}/>
          <button className="btn ghost" onClick={changePin}>Cambiar</button>
        </div>
      </Section>
      {saved&&<div style={{color:T.teal,fontSize:12,textAlign:"center"}}>{saved}</div>}
      <div style={{fontSize:11,color:T.dim,marginTop:10,lineHeight:1.6}}>
        Para compartir: publica esta app con el botón <b>Compartir/Publicar</b> del artefacto y entrega el enlace + un código.
        Esta puerta es un control ligero de comunidad; para producción real usa el backend incluido con autenticación de verdad.
      </div>
      </>)}
    </div>
  );
}

/* ============================ APP (con puerta de miembros) ============================ */
function App(){
  const [auth,setAuth]=useState(null); // null=gate, {admin:bool}
  const lock=async()=>{ try{await window.storage.delete("session");}catch{} setAuth(null); };
  if(auth===null) return <Gate onEnter={(admin)=>setAuth({admin})}/>;
  return <MainApp admin={auth.admin} onLock={lock}/>;
}

ReactDOM.createRoot(document.getElementById('root')).render(React.createElement(App));
