function buildStats(lot){
  const lottery=LOTTERIES.find(l=>l.id===lot);
  const freq=new Array(100).fill(0); const lastSeen=new Array(100).fill(9999);
  const days=pastDays(120);
  days.forEach((d,idx)=>{
    const dk=dateKey(d);
    lottery.sessions.forEach(s=>{
      const r=resultFor(lot,dk,s);
      [parseInt(r.fijo2,10),...r.corridos.map(c=>parseInt(c,10))].forEach(n=>{
        freq[n]++; if(idx<lastSeen[n]) lastSeen[n]=idx;
      });
    });
  });
  const all=Array.from({length:100},(_,n)=>({n,f:freq[n],days:lastSeen[n]===9999?120:lastSeen[n]}));
  return {
    hot:[...all].sort((a,b)=>b.f-a.f).slice(0,8),
    cold:[...all].sort((a,b)=>a.f-b.f).slice(0,8),
    late:[...all].sort((a,b)=>b.days-a.days).slice(0,8),
    all,
  };
}

function aiContext(lot, realDraws){
  const lottery=LOTTERIES.find(l=>l.id===lot);
  const dk=dateKey(new Date());
  let lines="(sin resultados oficiales cargados — dile al cliente que mire la pestaña 🎱 Hoy)";
  if(realDraws && realDraws.length){
    const mine=realDraws.filter(d=>(d.board||"labolita")===lot);
    if(mine.length){
      lines=mine.map(d=>{
        const f3=String(d.pick3).padStart(3,"0"), f2=f3.slice(1);
        const p4=String(d.pick4).padStart(4,"0");
        return `${d.session} (${d.date}): fijo ${f3} (charada ${charadaOf(parseInt(f2,10))}), corridos ${p4.slice(0,2)},${p4.slice(2,4)}`;
      }).join(" | ");
    }
  }
  const st=buildStats(lot);
  return `Lotería ${lottery.name}, hoy ${dk}. RESULTADOS OFICIALES: ${lines}. Estadística (de demostración, solo orientativa) — Calientes: ${st.hot.slice(0,5).map(x=>x.n).join(",")}. Atrasados: ${st.late.slice(0,5).map(x=>`${x.n}(${x.days}d)`).join(",")}.`;
}

/* ============================ TAB: ESTUDIO HERMES ============================ */
function TabStudio({lot}){
  const dk=dateKey(new Date());
  const r=resultFor(lot,dk,LOTTERIES.find(l=>l.id===lot).sessions[0]);
  const n=parseInt(r.fijo2,10); const sym=charadaOf(n);
  const [prompt,setPrompt]=useState(`Estampa cubana vintage del número ${n===0?100:n}, "${sym}", estilo cartel de lotería habanera, dorado y azul noche`);
  const [busy,setBusy]=useState(false);
  const [out,setOut]=useState(null);
  const [log,setLog]=useState("");
  const generate=async()=>{
    if(busy) return;
    setBusy(true); setOut(null); setLog("Hermes está pintando tu número…");
    try{
      const data=await askClaude(
        [{role:"user",content:`Usa la herramienta de Higgsfield para generar UNA imagen (modelo nano_banana_pro, aspect_ratio 1:1, count 1) con este prompt: "${prompt}". Cuando termine, responde con la URL de la imagen.`}],
        {mcp:true, maxTokens:1500});
      const url=firstUrlIn(data);
      const tr=toolResultsOf(data);
      setLog(tr.length?"Listo. Arte exclusivo de Hermes ✔":textOf(data).slice(0,200));
      if(url) setOut(url);
      else if(!tr.length) setLog("El estudio está descansando ahora mismo. El dueño puede reactivarlo desde su panel.");
    }catch(e){
      setLog("Error de conexión con el estudio IA. Intenta de nuevo.");
    }
    setBusy(false);
  };
  return (
    <div className="fade-in">
      <Section title="Estudio Hermes" sub={`Arte del día — la charada de hoy es ${n===0?100:n}: ${sym}`}>
        <div className="card">
          <textarea className="input" rows={3} value={prompt} onChange={e=>setPrompt(e.target.value)} style={{resize:"none",marginBottom:10}}/>
          <button className="btn primary" style={{width:"100%"}} onClick={generate} disabled={busy}>
            {busy?"Hermes está creando…":"✨ Generar arte de mi número"}
          </button>
          {log && <div style={{fontSize:11,color:T.dim,marginTop:10}}>{log}</div>}
          {out && (
            <div style={{marginTop:12}}>
              <img src={out} alt="Arte generado" style={{width:"100%",borderRadius:14,border:`1px solid ${T.line}`}}
                   onError={e=>{e.target.style.display="none";}}/>
              <a href={out} target="_blank" rel="noreferrer" style={{display:"block",marginTop:8,fontSize:12,color:T.teal,wordBreak:"break-all"}}>{out}</a>
            </div>
          )}
        </div>
      </Section>
      <div style={{fontSize:11,color:T.dim}}>
        Arte exclusivo del estudio de Hermes para compartir tu número cada día.
      </div>
    </div>
  );
}

/* ============================ TAB: FORO (storage compartido) ============================ */
function TabForo(){
  const dk=dateKey(new Date());
  const KEY=`foro:${dk}`;
  const [items,setItems]=useState(null);
  const [text,setText]=useState("");
  const [alias,setAlias]=useState("");
  const [busy,setBusy]=useState(false);
  useEffect(()=>{(async()=>{
    try{const a=await window.storage.get("alias"); setAlias(a?.value||"");}catch{ setAlias(""); }
    try{const r=await window.storage.get(KEY,true); setItems(JSON.parse(r.value));}
    catch{ setItems([]); }
  })();},[]);
  const post=async()=>{
    const t=text.trim(); if(!t||busy) return;
    setBusy(true);
    let name=alias;
    if(!name){ name="Anónimo-"+Math.random().toString(16).slice(2,8);
      try{await window.storage.set("alias",name);}catch{} setAlias(name); }
    const next=[{u:name,t,ts:Date.now(),up:0},...(items||[])].slice(0,200);
    try{ await window.storage.set(KEY,JSON.stringify(next),true); setItems(next); setText(""); }
    catch{ /* keep text so the user can retry */ }
    setBusy(false);
  };
  const vote=async(i)=>{
    const next=items.map((x,idx)=>idx===i?{...x,up:(x.up||0)+1}:x);
    setItems(next);
    try{ await window.storage.set(KEY,JSON.stringify(next),true);}catch{}
  };
  return (
    <div className="fade-in">
      <Section title="Foro del día" sub="Comparte tu criterio con la comunidad · visible para todos">
        <div className="card" style={{marginBottom:12}}>
          <textarea className="input" rows={2} placeholder="Mi criterio para la noche: 17 y 65, por la luna…" value={text} onChange={e=>setText(e.target.value)} style={{resize:"none",marginBottom:10}}/>
          <button className="btn primary" style={{width:"100%"}} onClick={post} disabled={busy}>Publicar como {alias||"Anónimo"}</button>
        </div>
        {items===null && <div style={{color:T.dim,fontSize:13}}>Cargando comentarios…</div>}
        {items&&items.length===0 && <div className="card" style={{color:T.dim,fontSize:13,textAlign:"center"}}>Nadie ha comentado hoy. Sé el primero.</div>}
        <div style={{display:"grid",gap:8}}>
          {(items||[]).map((c,i)=>(
            <div key={i} className="card" style={{padding:"12px 14px"}}>
              <div style={{display:"flex",justifyContent:"space-between",fontSize:11,color:T.dim,marginBottom:6}}>
                <span style={{color:T.mango,fontWeight:600}}>{c.u}</span>
                <span>{new Date(c.ts).toLocaleTimeString("es-ES",{hour:"2-digit",minute:"2-digit"})}</span>
              </div>
              <div style={{fontSize:14,lineHeight:1.5,whiteSpace:"pre-wrap"}}>{c.t}</div>
              <button className="chip" style={{marginTop:8}} onClick={()=>vote(i)}>👍 {c.up||0}</button>
            </div>
          ))}
        </div>
      </Section>
      <div style={{fontSize:11,color:T.dim,textAlign:"center"}}>
        Comunidad informativa. Aquí no se hacen ni se aceptan apuestas.
      </div>
    </div>
  );
}
