function hashStr(s){let h=2166136261;for(let i=0;i<s.length;i++){h^=s.charCodeAt(i);h=Math.imul(h,16777619);}return h>>>0;}
function mulberry32(a){return function(){a|=0;a=(a+0x6D2B79F5)|0;let t=Math.imul(a^(a>>>15),1|a);t=(t+Math.imul(t^(t>>>7),61|t))^t;return((t^(t>>>14))>>>0)/4294967296;};}
function resultFor(lotId, dateKey, session){
  const rng = mulberry32(hashStr(`${lotId}|${dateKey}|${session}`));
  const d3 = () => String(Math.floor(rng()*1000)).padStart(3,"0");
  const d2 = () => String(Math.floor(rng()*100)).padStart(2,"0");
  const fijo3 = d3();
  const fijo2 = fijo3.slice(1);
  const c1 = d2(), c2 = d2();
  return {
    fijo3, fijo2, corridos:[c1,c2],
    parles:[[fijo2,c1],[fijo2,c2],[c1,c2]],
    candado:[fijo2,c1,c2],
  };
}

function nextDraw(lot){
  const now=new Date();
  for(let add=0;add<2;add++){
    for(const s of lot.sessions){
      const t=new Date(now); t.setDate(now.getDate()+add);
      const h=SESSION_TIMES[s]; t.setHours(Math.floor(h),Math.round((h%1)*60),0,0);
      if(t>now) return {session:s,at:t};
    }
  }
  return {session:lot.sessions[0],at:now};
}

    else{
      const y=new Date(today); y.setDate(today.getDate()-1);
      const dy=pickDraw(draws,lot,dateKey(y),sess);
      if(dy && lstate==="ok"){ pending=true; res=toRes(dy.pick3, String(dy.pick4).slice(0,2), String(dy.pick4).slice(2,4)); shownDate=dy.date; }
      else pending=true;
    }
  } else {
    res=resultFor(lot,dk,sess);
  }

  return (
    <div className="fade-in">
      <div className="scroll-x" style={{marginBottom:14}}>
        {LOTTERIES.map(l=>(
          <button key={l.id} className={"chip"+(lot===l.id?" on":"")} onClick={()=>setLot(l.id)}>{l.short}</button>
        ))}
      </div>
      <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:12}}>
        <div style={{display:"flex",alignItems:"center",gap:8,fontSize:12,color:T.dim}}>
          <span className="live-dot"/> {(isLive && (live?.realBoards||[]).includes(lot)) ? (live?.source==="ia" ? "EN VIVO · búsqueda web" : "EN VIVO · datos oficiales") : "DEMO · referencia"} · {fmtDate(today)}
        </div>
        {isLive&&(
          <button className="chip" onClick={load} disabled={lstate==="loading"}>
            {lstate==="loading"?"Actualizando…":"⟳ Refrescar"}
          </button>
        )}
      </div>
      <div className="card" style={{padding:"10px 14px",display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:12}}>
        <span style={{fontSize:12,color:T.dim}}>Próximo tiro · {nd.session}</span>
        <Countdown to={nd.at.getTime()}/>
      </div>
      <div className="scroll-x" style={{marginBottom:12}}>
        {lottery.sessions.map(s=>(
          <button key={s} className={"chip"+(sess===s?" on":"")} onClick={()=>setSess(s)}>{s}</button>
        ))}
      </div>
      {isLive&&lstate==="loading"&&!res&&<PendingHero session={sess}/>}
      {isLive&&lstate==="err"&&!res&&(
        <div className="card" style={{textAlign:"center",color:T.dim,fontSize:13,padding:24}}>
          La señal en vivo se interrumpió un momento. Toca <b>⟳ Refrescar</b> para reintentar.
        </div>
      )}
      {isLive&&pending&&res&&(
        <div style={{fontSize:11,color:T.mango,textAlign:"center",marginBottom:8}}>
          El tiro de hoy aún no cae — mostrando el último resultado oficial ({shownDate}).
        </div>
      )}
      {res&&<FijoHero res={res} session={sess}/>}
      {res&&<ResultDetail res={res}/>}
      {!isLive&&(
        <div style={{marginTop:12,fontSize:11,color:T.dim,textAlign:"center"}}>
          Tablero de demostración.
        </div>
      )}
    </div>
  );
}
