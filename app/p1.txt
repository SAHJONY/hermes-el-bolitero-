const HIGGSFIELD_MCP = { type:"url", url:"https://mcp.higgsfield.ai/mcp", name:"higgsfield" };
async function askClaude(messages, { system, mcp=false, web=false, maxTokens=1000 } = {}){
  let env={};
  try{ const r=await window.storage.get("env",false); env=JSON.parse(r.value); }catch{}
  const antKey=env.ANTHROPIC_API_KEY;
  const groqKey=env.GROQ_API_KEY;
  const groqModel=env.GROQ_MODEL||"llama-3.3-70b-versatile";
  const nimKey=env.NVIDIA_API_KEY, nimModel=env.NVIDIA_MODEL||"nvidia/llama-3.1-nemotron-70b-instruct";
  const nimBase=env.NVIDIA_BASE_URL||"https://integrate.api.nvidia.com/v1";

  // SEÑAL EN VIVO (números) -> proxy del servidor /api/numeros (búsqueda web SIN CORS)
  if(web){
    const userMsg = messages.map(m=>typeof m.content==="string"?m.content:"").join("\n");
    const r=await fetch("/api/numeros",{method:"POST",headers:{"Content-Type":"application/json"},
      body:JSON.stringify({ prompt:userMsg, maxTokens })});
    const d=await r.json().catch(()=>({}));
    if(!r.ok) throw new Error(d.error||("numeros-"+r.status));
    return { content:[{type:"text",text:d.text||""}] };
  }
  // ESTUDIO (Higgsfield MCP) -> Anthropic directo desde el navegador
  if(mcp){
    if(!antKey) throw new Error("sin-anthropic");
    const body={ model:"claude-sonnet-4-6", max_tokens:maxTokens, messages, mcp_servers:[HIGGSFIELD_MCP] };
    if(system) body.system=system;
    const r=await fetch("https://api.anthropic.com/v1/messages",{
      method:"POST",
      headers:{"Content-Type":"application/json","x-api-key":antKey,"anthropic-version":"2023-06-01","anthropic-dangerous-direct-browser-access":"true"},
      body:JSON.stringify(body)});
    return r.json();
  }
  // CEREBRO DE HERMES (chat + sueños): primero el proxy server-side /api/hermes,
  // donde la clave del modelo vive SOLO en el servidor. Si el proxy no está
  // configurado (503) o falla, cae a una clave guardada en el panel de Admin:
  // Groq directo desde el navegador y, como último respaldo, NVIDIA directo.
  const asOpenAI = (base, key, model) => fetch(`${base}/chat/completions`,{
      method:"POST",
      headers:{"Content-Type":"application/json","Authorization":`Bearer ${key}`},
      body:JSON.stringify({ model, max_tokens:Math.min(maxTokens,800), temperature:0.6,
        messages: system?[{role:"system",content:system},...messages]:messages })});

  // 1) Proxy server-side (la clave nunca sale del servidor).
  try{
    const r=await fetch("/api/hermes",{method:"POST",headers:{"Content-Type":"application/json"},
      body:JSON.stringify({ messages, system, max_tokens:Math.min(maxTokens,800) })});
    if(r.ok){ const d=await r.json(); if(d&&Array.isArray(d.content)) return { content:d.content }; }
  }catch(e){}
  // 2) Respaldo: Groq directo desde el navegador (clave guardada en Admin).
  if(groqKey){
    const r=await asOpenAI("https://api.groq.com/openai/v1", groqKey, groqModel);
    const d=await r.json();
    if(!r.ok) throw new Error(d.error?.message||("groq-"+r.status));
    return { content:[{type:"text",text:d.choices?.[0]?.message?.content ?? ""}] };
  }
  // 3) Último respaldo: NVIDIA directo (puede fallar por CORS en navegador)
  if(nimKey){
    const r=await asOpenAI(nimBase, nimKey, nimModel);
    const d=await r.json();
    return { content:[{type:"text",text:d.choices?.[0]?.message?.content ?? ""}] };
  }
  throw new Error("sin-cerebro");
}

function parseJsonLoose(t){
  try{ return JSON.parse(t.replace(/```json|```/g,"").trim()); }

}

      const j=parseJsonLoose(textOf(data));
      if(j&&Array.isArray(j.draws)&&j.draws.length){
        aiDraws=j.draws.filter(d=>missing.includes(d.board));
        aiBoards=[...new Set(aiDraws.map(d=>d.board))];
      }
    }catch(e){}
  }

  // Merge oficial + IA, dedupe por board|date|session (lo oficial siempre gana).
  const seen=new Set(officialDraws.map(d=>`${d.board}|${d.date}|${d.session}`));

function TabHoy({lot,setLot,go}){
  const lottery = LOTTERIES.find(l=>l.id===lot);
  const isLive = !!lottery.live;
  const [sess,setSess]=useState(lottery.sessions[0]);
  const [live,setLive]=useState(null);     // {ts,draws}
  const [lstate,setLstate]=useState("idle"); // idle|loading|ok|err
  const [favs,setFavs]=useState([]);
  useLang();
  useEffect(()=>{(async()=>{ const r=await sGet("favs",false); if(Array.isArray(r)) setFavs(r); })();},[]);
  const favSet=new Set(favs);
  // Personalización: las loterías favoritas suben al frente del carrusel.
  const orderedLots=[...LOTTERIES].sort((a,b)=>(favSet.has(b.id)?1:0)-(favSet.has(a.id)?1:0));
  const toggleFav=async()=>{ const next=favSet.has(lot)?favs.filter(x=>x!==lot):[...favs,lot]; setFavs(next); await sSet("favs",next,false); };
  useEffect(()=>{setSess(lottery.sessions[0]);},[lot]);
  const load=async()=>{
    setLstate("loading");
    try{ setLive(await fetchLiveDraws()); setLstate("ok"); }
    catch{ setLstate("err"); }
  };
  useEffect(()=>{ if(isLive){ load(); const i=setInterval(load, LIVE_TTL_MIN*60*1000); return ()=>clearInterval(i);} },[lot]);
  const today=new Date(); const dk=dateKey(today);
  const nd=nextDraw(lottery);

  let res=null, pending=false, shownDate=null;
  if(isLive){
    const draws=live?.draws||[];
    const d=pickDraw(draws,lot,dk,sess);
    if(d){ res=toRes(d.pick3, String(d.pick4).slice(0,2), String(d.pick4).slice(2,4)); shownDate=d.date; }
    else{
      const y=new Date(today); y.setDate(today.getDate()-1);
      const dy=pickDraw(draws,lot,dateKey(y),sess);
      if(dy && lstate==="ok"){ pending=true; res=toRes(dy.pick3, String(dy.pick4).slice(0,2), String(dy.pick4).slice(2,4)); shownDate=dy.date; }
      else pending=true;
    }
  } else {
    res=resultFor(lot,dk,sess);
  }

  return (
    <div className="fade-in">
      <CasinoHero go={go}/>
      <div className="scroll-x" style={{marginBottom:14}}>
        {orderedLots.map(l=>(
          <button key={l.id} className={"chip"+(lot===l.id?" on":"")} onClick={()=>setLot(l.id)}>
            {favSet.has(l.id)?"★ ":""}{l.short}
          </button>
        ))}
      </div>
      <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:12,gap:8}}>
        <div style={{display:"flex",alignItems:"center",gap:8,fontSize:12,color:T.dim,minWidth:0}}>
          <span className="live-dot"/> <span style={{overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{(isLive && (live?.realBoards||[]).includes(lot)) ? t("today.live_official") : (isLive && (live?.aiBoards||[]).includes(lot)) ? t("today.live_web") : t("today.demo")} · {fmtDate(today)}</span>
        </div>
        <div style={{display:"flex",gap:6,flexShrink:0}}>
          <button className="chip" onClick={toggleFav} title={t("today.fav")} aria-label={t("today.fav")} aria-pressed={favSet.has(lot)} style={favSet.has(lot)?{borderColor:T.mango,color:T.mango}:undefined}>{favSet.has(lot)?"★":"☆"}</button>
          {isLive&&(
            <button className="chip" onClick={load} disabled={lstate==="loading"}>
              {lstate==="loading"?t("today.updating"):t("today.refresh")}
            </button>
          )}
        </div>
      </div>
      <div className="card" style={{padding:"10px 14px",display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:12}}>
        <span style={{fontSize:12,color:T.dim}}>{t("today.next")} · {nd.session}</span>
        <Countdown to={nd.at.getTime()}/>
      </div>
      <div className="scroll-x" style={{marginBottom:12}}>
        {lottery.sessions.map(s=>(
          <button key={s} className={"chip"+(sess===s?" on":"")} onClick={()=>setSess(s)}>{s}</button>
        ))}
      </div>
      {isLive&&lstate==="loading"&&!res&&<PendingHero session={sess}/>}
      {isLive&&lstate==="err"&&!res&&(
        <div className="card" style={{textAlign:"center",color:T.dim,fontSize:13,padding:24}}>
          La señal en vivo se interrumpió un momento. Toca <b>⟳ Refrescar</b> para reintentar.
        </div>
      )}
      {isLive&&pending&&res&&(
        <div style={{fontSize:11,color:T.mango,textAlign:"center",marginBottom:8}}>
          El tiro de hoy aún no cae — mostrando el último resultado oficial ({shownDate}).
        </div>
      )}
      {res&&<FijoHero res={res} session={sess}/>}
      {res&&<ResultDetail res={res}/>}
      {isLive&&res&&(live?.aiBoards||[]).includes(lot)&&(
        <div style={{marginTop:10,fontSize:11,color:T.mango,textAlign:"center",lineHeight:1.5}}>
          🔎 Números aproximados de búsqueda web — no es el resultado oficial. Confírmalos en la lotería oficial de tu estado.
        </div>
      )}
      {!isLive&&(
        <div style={{marginTop:12,fontSize:11,color:T.dim,textAlign:"center"}}>
          Tablero de demostración.
        </div>
      )}
    </div>
  );
}
